1//! The text and tags of the HTML embedded in Markdown, from html5gum's
2//! tokenizer. Nothing here builds a tree.
3
4use html5gum::{DefaultEmitter, Token as Html, Tokenizer};
5use ratatui_core::style::Style;

7use super::{BULLETS, Capture, Kind, Open, Renderer};
8use crate::wrap::Piece;

11/// Text or a tag, read from HTML.
12#[derive(Debug, Clone, PartialEq)]
13pub(crate) enum Token {
14    Text(String),
15    Open {
16        name: String,
17        attrs: Vec<(String, String)>,
18    },
19    Close(String),
20}
21
22impl Token {
23    /// Splits HTML into text and tags. Comments are dropped, along with the
24    /// content of [`SKIPPED`] elements, though their opening tag is still
25    /// returned so its attributes can be used.
26    pub fn parse(html: &str) -> Vec<Self> {
27        let mut emitter = DefaultEmitter::default();
28        // Read the insides of elements like `<script>` as text, like a browser.
29        emitter.naively_switch_states(true);
30        let mut tokens: Vec<Self> = Vec::new();
31        let mut skipping: Option<String> = None;
32        for token in Tokenizer::new_with_emitter(html, emitter).flatten() {
33            match token {
34                Html::StartTag(tag) if skipping.is_none() => {
35                    let name = string(&tag.name);
36                    if SKIPPED.contains(&name.as_str()) && !tag.self_closing {
37                        skipping = Some(name.clone());
38                    }
39                    let attrs = tag
40                        .attributes
41                        .iter()
42                        .map(|(k, v)| (string(k), string(v)))
43                        .collect();
44                    tokens.push(Token::Open { name, attrs });
45                }
46                Html::EndTag(tag) => {
47                    let name = string(&tag.name);
48                    if skipping.is_none() {
49                        tokens.push(Token::Close(name));
50                    } else if skipping.as_ref() == Some(&name) {
51                        skipping = None;
52                    }
53                }
54                Html::String(text) if skipping.is_none() => {
55                    // The tokenizer can hand text over in pieces.
56                    if let Some(Token::Text(last)) = tokens.last_mut() {
57                        last.push_str(&string(&text));
58                    } else {
59                        tokens.push(Token::Text(string(&text)));
60                    }
61                }
62                _ => {}
63            }
64        }
65        tokens
66    }
67
68    /// The value of an opening tag's attribute `key`.
69    pub fn attr(&self, key: &str) -> Option<&str> {
70        match self {
71            Token::Open { attrs, .. } => attrs
72                .iter()
73                .find(|(k, _)| k == key)
74                .map(|(_, v)| v.as_str()),
75            _ => None,
76        }
77    }
78}

80/// Elements that never have content, or a closing tag.
81const VOID: &[&str] = &[
82    "area", "base", "br", "col", "embed", "hr", "img", "input", "link", "meta", "source", "track",
83    "wbr",
84];
85
86/// Elements that start a new line, and end one.
87const BLOCKS: &[&str] = &[
88    "p",
89    "div",
90    "ul",
91    "ol",
92    "tr",
93    "table",
94    "blockquote",
95    "section",
96    "article",
97    "header",
98    "footer",
99    "figure",
100    "dl",
101    "dt",
102    "dd",
103];
104
105/// Headings, by level.
106const HEADINGS: [&str; 6] = ["h1", "h2", "h3", "h4", "h5", "h6"];
107
108/// Elements whose content is never shown as text.
109const SKIPPED: &[&str] = &["script", "style", "svg", "template", "head"];
110
111fn string(bytes: &[u8]) -> String {
112    String::from_utf8_lossy(bytes).into_owned()
113}

115impl Renderer<'_> {
116    /// Draws an HTML block once it's been read whole.
117    pub(super) fn html_block(&mut self, html: &str) {
118        let before = self.writer.lines_written();
119        self.html_tokens(html);
120        // A `<pre>` left open ends with the block.
121        if matches!(self.capture, Some(Capture::Pre(_))) {
122            self.draw_capture();
123        }
124        // So do HTML styles and links.
125        while matches!(
126            self.open.last(),
127            Some(Open {
128                kind: Kind::Html(_),
129                ..
130            })
131        ) {
132            self.open.pop();
133        }
134        // Lots of HTML, like a closing `</div>`, shows nothing, and
135        // shouldn't leave a blank line behind.
136        if self.writer.inline_is_blank() && self.writer.lines_written() == before {
137            self.writer.take_inline();
138        } else {
139            self.writer.end_block();
140        }
141    }
142
143    /// Reads the text and links out of a chunk of HTML.
144    pub(super) fn html_tokens(&mut self, html: &str) {
145        for token in Token::parse(html) {
146            // Inside a `<pre>`, only its text and its end matter.
147            if let Some(Capture::Pre(pre)) = &mut self.capture {
148                match &token {
149                    Token::Text(text) => pre.push_str(text),
150                    Token::Close(name) if name == "pre" => {
151                        self.draw_capture();
152                        self.close_class(name);
153                    }
154                    _ => {}
155                }
156                continue;
157            }
158            match &token {
159                Token::Text(text) => self.text(text.clone()),
160                Token::Open { name, .. } => {
161                    self.open_class(name, token.attr("class"));
162                    self.open_tag(&token, name);
163                }
164                Token::Close(name) => {
165                    self.close_tag(name);
166                    self.close_class(name);
167                }
168            }
169        }
170    }
171
172    /// Opens a [`Kind::Class`] for an element with classes the theme styles,
173    /// or for one with the same name inside a styled element, so the closing
174    /// tags pair up with the right one.
175    fn open_class(&mut self, name: &str, classes: Option<&str>) {
176        if VOID.contains(&name) {
177            return;
178        }
179        let style = classes.and_then(|c| self.theme.class_style(c.split_whitespace()));
180        let kind = Kind::Class {
181            name: name.to_owned(),
182            background: style.and_then(|style| style.bg),
183        };
184        if style.is_some() || self.is_open(&kind) {
185            self.open(kind, style.unwrap_or_default());
186        }
187    }
188
189    fn close_class(&mut self, name: &str) {
190        let kind = Kind::Class {
191            name: name.to_owned(),
192            background: None,
193        };
194        self.close(&kind);
195    }
196
197    fn open_tag(&mut self, token: &Token, name: &str) {
198        match name {
199            "a" => {
200                if let Some(href) = token.attr("href") {
201                    self.open_link(Kind::html(name), href);
202                }
203            }
204            "br" => self.writer.line_break(),
205            "pre" => {
206                self.writer.flush();
207                self.capture = Some(Capture::Pre(String::new()));
208            }
209            "li" => {
210                // Like a Markdown item, wrapped lines hang under
211                // the text, not the bullet.
212                self.writer.flush();
213                let bullet = Piece::new(BULLETS[0], self.theme.list_marker);
214                let prefix = Prefix::hanging(bullet, self.theme.text);
215                self.open_prefixed(Kind::HtmlItem, prefix, Style::new());
216            }
217            _ if BLOCKS.contains(&name) => self.writer.hard_break(),
218            "hr" => {
219                self.writer.hard_break();
220                self.text("───");
221                self.writer.line_break();
222            }
223            _ if HEADINGS.contains(&name) => {
224                // A blank line before a heading, as in Markdown.
225                self.writer.hard_break();
226                if self.writer.inline().is_empty() {
227                    self.writer.set_gap(true);
228                } else {
229                    self.writer.line_break();
230                }
231                let level = HEADINGS.iter().position(|h| *h == name).unwrap_or(0);
232                self.heading_marker(level);
233                self.open(Kind::html(name), self.theme.headings[level]);
234            }
235            "img" => {
236                let src = token.attr("src").unwrap_or("");
237                let alt = token.attr("alt").unwrap_or("");
238                self.image(src, alt);
239            }
240            "svg" => {
241                if let Some(label) = token.attr("aria-label") {
242                    let style = self.style().patch(self.theme.image);
243                    self.piece(format!("[figure: {label}]"), style);
244                }
245            }
246            _ => {
247                if let Some(style) = self.tag_style(name) {
248                    self.open(Kind::html(name), style);
249                }
250            }
251        }
252    }
253
254    /// The style of text in an element that formats it, like `<strong>`.
255    fn tag_style(&self, name: &str) -> Option<Style> {
256        Some(match name {
257            "strong" | "b" => self.theme.strong,
258            "em" | "i" | "cite" => self.theme.emphasis,
259            "code" | "kbd" | "tt" | "samp" => self.theme.code,
260            "s" | "del" | "strike" => self.theme.strikethrough,
261            "small" | "sub" | "sup" => self.theme.small,
262            _ => return None,
263        })
264    }
265
266    fn close_tag(&mut self, name: &str) {
267        match name {
268            "li" if self.is_open(&Kind::HtmlItem) => {
269                self.writer.flush();
270                self.close(&Kind::HtmlItem);
271            }
272            _ if name == "li" || BLOCKS.contains(&name) => self.writer.hard_break(),
273            "td" | "th" => self.text(" "),
274            _ => {
275                // Only when it's the innermost, as HTML is often
276                // left unclosed.
277                if matches!(self.open.last(), Some(o) if o.kind.is_like(&Kind::html(name))) {
278                    self.open.pop();
279                }
280                if HEADINGS.contains(&name) {
281                    self.writer.hard_break();
282                }
283            }
284        }
285    }
286
287    /// Draws the text of an HTML `<pre>` like a code block.
288    pub(super) fn pre(&mut self, text: &str) {
289        // Like a browser, skip one newline right after `<pre>`.
290        let text = text.strip_prefix('\n').unwrap_or(text);
291        self.code_block("", text);
292    }
293}

295#[cfg(test)]
296mod tests {
297    use super::*;
298
299    #[test]
300    fn links_and_text() {
301        let tokens = Token::parse(r#"<li><a href="/x?a=1&amp;b=2">Hi &amp; bye</a></li>"#);
302        assert_eq!(tokens[1].attr("href"), Some("/x?a=1&b=2"));
303        assert_eq!(tokens[2], Token::Text("Hi & bye".into()));
304        assert_eq!(tokens[3], Token::Close("a".into()));
305    }
306
307    #[test]
308    fn skips_svg_content_and_comments() {
309        let tokens = Token::parse(r#"a<!-- no --><svg aria-label="fig"><text>x</text></svg>b"#);
310        assert_eq!(tokens.len(), 3);
311        assert_eq!(tokens[1].attr("aria-label"), Some("fig"));
312        assert_eq!(tokens[2], Token::Text("b".into()));
313    }
314
315    #[test]
316    fn multibyte_text() {
317        assert_eq!(
318            Token::parse("λ<b>"),
319            [
320                Token::Text("λ".into()),
321                Token::Open {
322                    name: "b".into(),
323                    attrs: vec![]
324                }
325            ]
326        );
327    }
328
329    #[test]
330    fn stray_angle_brackets_are_text() {
331        let tokens = Token::parse("1 < 2 > 0");
332        let text: String = tokens
333            .iter()
334            .map(|t| match t {
335                Token::Text(s) => s.as_str(),
336                _ => "",
337            })
338            .collect();
339        assert_eq!(text, "1 < 2 > 0");
340    }
