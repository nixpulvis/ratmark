1//! Scrolling with a wheel, a trackpad or a finger, a cell at a time.
2//!
3//! Each gesture only goes one way, down the page, or sideways through the
4//! wide code block under the pointer, or where a finger first touched, so a
5//! swipe that's a little diagonal doesn't do both. Hosts that report
6//! movement in pixels, like a browser, move the page as far as the hand
7//! does, a line or a column at a time once one's worth adds up, and a flick
8//! keeps the page gliding for a bit, slowing down, like a page in the
9//! browser does.
10//!
11//! [`Gestures`] does the arithmetic, from pixels and timestamps, and hands
12//! each whole cell of movement to a callback as a [`Step`], which
13//! [`Step::apply`] carries out on a [`MarkdownState`]. Each host feeds it
14//! its own events:
15//!
16//! - In a terminal, [`terminal::Wheel`] takes the wheel's ticks, like
17//!   crossterm's `MouseEventKind::ScrollDown`.
18//! - In a browser, the `web` feature's `web::listen` follows the wheel and
19//!   touch events over the element a view is drawn in, like Ratzilla's.
20//! - Anywhere else that reports movement in pixels, call [`Gestures`]'
21//!   methods from its events, and [`Gestures::glide`] once a frame after
22//!   [`Gestures::touch_end`] says a touch was a flick.
23//!
24//! ```
25//! use ratmark::MarkdownState;
26//! use ratmark::gesture::Gestures;
27//! # use ratatui_core::{buffer::Buffer, layout::Rect, widgets::StatefulWidget};
28//!
29//! let markdown = (1..=20).map(|n| format!("{n}\n\n")).collect::<String>();
30//! let mut state = MarkdownState::new(markdown);
31//! # let area = Rect::new(0, 0, 10, 5);
32//! # ratmark::MarkdownView::new(&Default::default())
33//! #     .render(area, &mut Buffer::empty(area), &mut state);
34//! // Cells 8 pixels wide and 16 tall.
35//! let mut gestures = Gestures::new(8.0, 16.0);
36//!
37//! // Half a line doesn't move yet, the rest of a line and a half does.
38//! gestures.wheel((0.0, 0.0), (0.0, 8.0), 0.0, |step| step.apply(&mut state));
39//! assert_eq!(state.scroll(), 0);
40//! gestures.wheel((0.0, 0.0), (0.0, 24.0), 10.0, |step| step.apply(&mut state));
41//! assert_eq!(state.scroll(), 2);
42//! ```

51/// Slowest a flick glides, in pixels a millisecond.
52const MIN_GLIDE: f64 = 0.05;
53
54/// How far a finger goes before its touch picks a way to move, in pixels.
55const LOCK_DISTANCE: f64 = 6.0;
56
57/// How long between wheel events before the next starts a new gesture, in
58/// milliseconds.
59const WHEEL_GESTURE: f64 = 200.0;
60
61/// How long a glide's first frame is taken to be, in milliseconds.
62const FRAME: f64 = 16.0;
63
64/// The way a gesture moves the view. A gesture only ever moves one way.
65#[derive(Debug, Clone, Copy, PartialEq, Eq)]
66pub enum Axis {
67    /// Up and down the page, scrolling.
68    Vertical,
69    /// Left and right, shifting wide code.
70    Horizontal,
71}
72
73impl Axis {
74    /// The way a move of `x`, `y` pixels mostly goes.
75    fn of(x: f64, y: f64) -> Self {
76        if x.abs() > y.abs() {
77            Axis::Horizontal
78        } else {
79            Axis::Vertical
80        }
81    }
82
83    /// The part of a move of `x`, `y` pixels that goes this way.
84    fn part(self, x: f64, y: f64) -> f64 {
85        match self {
86            Axis::Vertical => y,
87            Axis::Horizontal => x,
88        }
89    }
90}

92/// A gesture's move of whole cells.
93#[derive(Debug, Clone, Copy, PartialEq, Eq)]
94pub struct Step {
95    /// Which way it goes.
96    pub axis: Axis,
97    /// How many lines down, or columns right, it goes. Negative goes up, or
98    /// left.
99    pub cells: isize,
100    /// The column of the cell the gesture is over: the pointer for a wheel,
101    /// or where the finger came down for a touch.
102    pub column: u16,
103    /// The row of the cell the gesture is over.
104    pub row: u16,
105}
106
107impl Step {
108    /// Scrolls `state`, or shifts the wide code block under the gesture.
109    /// Returns whether it moved, which it doesn't at the end, or sideways
110    /// away from wide code.
111    pub fn apply(self, state: &mut MarkdownState) -> bool {
112        match self.axis {
113            Axis::Vertical => state.scroll_by(self.cells),
114            Axis::Horizontal => state.shift_at(self.column, self.row, self.cells),
115        }
116    }
117}

119/// Turns wheel and touch movement, in pixels, into [`Step`]s of whole
120/// cells, keeping the part of a cell left over for the gesture's next
121/// event.
122///
123/// Positions are in pixels from the view's top left corner, and times in
124/// milliseconds from any fixed point, like a browser event's `timeStamp`.
125/// Each method that moves takes a `step` callback, which carries out a
126/// [`Step`] and returns whether it moved, which it doesn't at the end.
127#[derive(Debug, Clone)]
128pub struct Gestures {
129    /// How wide and tall the view's cells are, in pixels.
130    cell: (f64, f64),
131    /// The part of a cell left over, down and sideways.
132    carried: (f64, f64),
133    /// Where the pointer, or the finger, last was.
134    pointer: (f64, f64),
135    /// The way the wheel gesture goes, and when its last event came, while
136    /// it hasn't run into the end.
137    wheel: Option<(Axis, f64)>,
138    touch: Touch,
139}
140
141impl Gestures {
142    /// Gestures over a view with cells `width` by `height` pixels.
143    pub fn new(width: f64, height: f64) -> Self {
144        Self {
145            cell: (width, height),
146            carried: (0.0, 0.0),
147            pointer: (0.0, 0.0),
148            wheel: None,
149            touch: Touch::default(),
150        }
151    }
152
153    /// A wheel event at `at`, going `delta` pixels right and down, at
154    /// `time`. A wheel gesture, like a swipe on a trackpad, goes the way its
155    /// first event mostly goes, until a pause, or until it runs into the
156    /// end, like a sideways flick's momentum at the end of the code.
157    pub fn wheel(
158        &mut self,
159        at: (f64, f64),
160        delta: (f64, f64),
161        time: f64,
162        step: impl FnOnce(Step) -> bool,
163    ) {
164        self.pointer = at;
165        let (x, y) = delta;
166        let axis = match self.wheel {
167            Some((axis, last)) if time - last < WHEEL_GESTURE => axis,
168            _ => {
169                let axis = Axis::of(x, y);
170                self.start(axis);
171                axis
172            }
173        };
174        let moved = self.move_by(axis, axis.part(x, y), step);
175        self.wheel = moved.then_some((axis, time));
176    }
177
178    /// A finger coming down at `at`, at `time`, catching any glide.
179    pub fn touch_start(&mut self, at: (f64, f64), time: f64) {
180        self.pointer = at;
181        self.touch = Touch {
182            start: Some(at),
183            last: Some(at),
184            at: time,
185            ..Touch::default()
186        };
187    }
188
189    /// The finger moving to `at`, at `time`, dragging the page along. It
190    /// doesn't move until it's gone far enough to pick which way it goes.
191    pub fn touch_move(&mut self, at: (f64, f64), time: f64, step: impl FnOnce(Step) -> bool) {
192        let touch = &mut self.touch;
193        let (Some(start), Some(last)) = (touch.start, touch.last) else {
194            return;
195        };
196        let axis = match touch.axis {
197            Some(axis) => axis,
198            None => {
199                let (x, y) = (at.0 - start.0, at.1 - start.1);
200                if x.abs().max(y.abs()) < LOCK_DISTANCE {
201                    return;
202                }
203                let axis = *touch.axis.insert(Axis::of(x, y));
204                self.start(axis);
205                axis
206            }
207        };
208        let touch = &mut self.touch;
209        touch.last = Some(at);
210        let pixels = axis.part(last.0 - at.0, last.1 - at.1);
211        let ms = (time - touch.at).max(1.0);
212        // Smoothed, so one jittery event doesn't decide a flick.
213        touch.speed = 0.8 * pixels / ms + 0.2 * touch.speed;
214        touch.at = time;
215        let _ = self.move_by(axis, pixels, step);
216    }
217
218    /// The finger lifting. Returns whether it was a flick, which glides on
219    /// with [`glide`](Self::glide) once a frame.
220    pub fn touch_end(&mut self) -> bool {
221        let touch = &mut self.touch;
222        touch.start = None;
223        touch.last = None;
224        touch.gliding = touch.axis.take().filter(|_| touch.speed.abs() > MIN_GLIDE);
225        touch.glided = None;
226        touch.gliding.is_some()
227    }
228
229    /// Keeps a flick going, for a frame drawn at `time`. Returns whether
230    /// it's still gliding, until it slows to a stop, runs into the end, or
231    /// another touch catches it.
232    pub fn glide(&mut self, time: f64, step: impl FnOnce(Step) -> bool) -> bool {
233        let touch = &mut self.touch;
234        let Some(axis) = touch.gliding else {
235            return false;
236        };
237        let ms = touch
238            .glided
239            .map_or(FRAME, |last| (time - last).clamp(1.0, 100.0));
240        touch.glided = Some(time);
241        let pixels = touch.speed * ms;
242        // About 5% slower every 16 milliseconds.
243        touch.speed *= 0.95_f64.powf(ms / FRAME);
244        if touch.speed.abs() <= MIN_GLIDE {
245            touch.gliding = None;
246        }
247        if !self.move_by(axis, pixels, step) {
248            self.touch.gliding = None;
249        }
250        self.touch.gliding.is_some()
251    }
252
253    /// Starts a gesture going `axis`, dropping what the last one left over,
254    /// so this one doesn't start part of a cell in.
255    fn start(&mut self, axis: Axis) {
256        *self.carried_mut(axis) = 0.0;
257    }
258
259    /// Moves `pixels` further along `axis`, and says whether the view
260    /// moved, which it doesn't at the end.
261    fn move_by(&mut self, axis: Axis, pixels: f64, step: impl FnOnce(Step) -> bool) -> bool {
262        let size = axis.part(self.cell.0, self.cell.1);
263        let carried = self.carried_mut(axis);
264        let total = *carried + pixels;
265        let cells = (total / size).trunc();
266        *carried = total - cells * size;
267        if cells == 0.0 {
268            return true;
269        }
270        let at = |pixels: f64, size: f64| (pixels / size).max(0.0) as u16;
271        step(Step {
272            axis,
273            cells: cells as isize,
274            column: at(self.pointer.0, self.cell.0),
275            row: at(self.pointer.1, self.cell.1),
276        })
277    }
278
279    fn carried_mut(&mut self, axis: Axis) -> &mut f64 {
280        match axis {
281            Axis::Vertical => &mut self.carried.1,
282            Axis::Horizontal => &mut self.carried.0,
283        }
284    }
285}

287/// A finger on the screen, and how fast it was going.
288#[derive(Debug, Clone, Default)]
289struct Touch {
290    /// Where the finger came down, and where it was last.
291    start: Option<(f64, f64)>,
292    last: Option<(f64, f64)>,
293    /// The way this touch moves the view, once it's gone far enough to tell.
294    axis: Option<Axis>,
295    /// When the last touch event came.
296    at: f64,
297    /// In pixels a millisecond, down or right when positive.
298    speed: f64,
299    /// The way a flick is gliding, and when its last frame was.
300    gliding: Option<Axis>,
301    glided: Option<f64>,
302}

308    /// A callback recording the steps it takes, which move until there are
309    /// `end` of them.
310    fn record(steps: &mut Vec<Step>, end: usize) -> impl FnOnce(Step) -> bool + '_ {
311        move |step| {
312            steps.push(step);
313            steps.len() < end
314        }
315    }
316
317    fn cells(steps: &[Step]) -> Vec<(Axis, isize)> {
318        steps.iter().map(|s| (s.axis, s.cells)).collect()
319    }

321    #[test]
322    fn moves_a_cell_at_a_time() {
323        let mut gestures = Gestures::new(10.0, 10.0);
324        let mut steps = Vec::new();
325        for (pixels, time) in [(4.0, 0.0), (4.0, 10.0), (25.0, 20.0), (-30.0, 30.0)] {
326            gestures.wheel((0.0, 0.0), (0.0, pixels), time, record(&mut steps, 10));
327        }
328        // 8 pixels isn't a cell yet, 33 is 3 with 3 over, and -27 is -2.
329        assert_eq!(cells(&steps), [(Axis::Vertical, 3), (Axis::Vertical, -2)]);
330        // After a pause a new gesture starts from nothing, rather than the -7
331        // left over, so two 5s make a cell.
332        gestures.wheel((0.0, 0.0), (0.0, 5.0), 500.0, record(&mut steps, 10));
333        gestures.wheel((0.0, 0.0), (0.0, 5.0), 510.0, record(&mut steps, 10));
334        assert_eq!(steps.len(), 3);
335    }

337    #[test]
338    fn wheels_move_one_way_until_the_end() {
339        let mut gestures = Gestures::new(10.0, 10.0);
340        let mut steps = Vec::new();
341        gestures.wheel((25.0, 35.0), (20.0, 5.0), 0.0, record(&mut steps, 10));
342        // Still sideways, though now it's mostly down.
343        gestures.wheel((25.0, 35.0), (10.0, 40.0), 50.0, record(&mut steps, 10));
344        assert_eq!(
345            cells(&steps),
346            [(Axis::Horizontal, 2), (Axis::Horizontal, 1)]
347        );
348        // Over the cell at the pointer.
349        assert_eq!((steps[0].column, steps[0].row), (2, 3));
350        // Running into the end frees the next event to go the other way.
351        gestures.wheel((0.0, 0.0), (10.0, 0.0), 60.0, record(&mut steps, 3));
352        gestures.wheel((0.0, 0.0), (0.0, 10.0), 70.0, record(&mut steps, 10));
353        assert_eq!(steps[3].axis, Axis::Vertical);
354    }

356    #[test]
357    fn touches_move_one_way() {
358        let mut gestures = Gestures::new(1.0, 1.0);
359        let mut steps = Vec::new();
360        gestures.touch_start((100.0, 100.0), 0.0);
361        // Too short a move to tell which way.
362        gestures.touch_move((103.0, 102.0), 10.0, record(&mut steps, 10));
363        assert!(steps.is_empty());
364        gestures.touch_move((90.0, 104.0), 20.0, record(&mut steps, 10));
365        // Still sideways, though now it's mostly down.
366        gestures.touch_move((88.0, 140.0), 30.0, record(&mut steps, 10));
367        assert_eq!(
368            cells(&steps),
369            [(Axis::Horizontal, 10), (Axis::Horizontal, 2)]
370        );
371        // Over where the finger came down.
372        assert_eq!((steps[1].column, steps[1].row), (100, 100));
373        // Held still before lifting, so it doesn't glide.
374        gestures.touch_move((88.0, 140.0), 500.0, record(&mut steps, 10));
375        assert!(!gestures.touch_end());
376    }

378    /// A finger flicking `gestures` up the page, and lifting.
379    fn flick(gestures: &mut Gestures) -> bool {
380        gestures.touch_start((100.0, 300.0), 0.0);
381        for (i, y) in [280.0, 240.0, 200.0].into_iter().enumerate() {
382            gestures.touch_move((100.0, y), 10.0 * (i + 1) as f64, |_| true);
383        }
384        gestures.touch_end()
385    }

387    #[test]
388    fn flicks_glide_until_they_stop() {
389        let mut gestures = Gestures::new(1.0, 1.0);
390        let mut steps = Vec::new();
391        assert!(flick(&mut gestures));
392        let mut time = 100.0;
393        while gestures.glide(time, record(&mut steps, usize::MAX)) {
394            time += 16.0;
395        }
396        // It glides on down, then slows to a stop.
397        assert!(steps.len() > 1);
398        assert!(
399            steps
400                .iter()
401                .all(|s| s.axis == Axis::Vertical && s.cells > 0)
402        );
403        assert!(steps[0].cells > steps[steps.len() - 1].cells);
404        assert!(!gestures.glide(time + 16.0, record(&mut steps, usize::MAX)));
405
406        // It stops at the end.
407        assert!(flick(&mut gestures));
408        assert!(!gestures.glide(100.0, |_| false));
409
410        // Or when another touch catches it.
411        assert!(flick(&mut gestures));
412        gestures.touch_start((100.0, 300.0), 200.0);
413        assert!(!gestures.glide(216.0, |_| panic!("caught glides on")));
414    }
