14/// Starts following the wheel and touches over the view drawn in `screen`,
15/// with cells `cell` CSS pixels wide and tall, handing each [`Step`] to
16/// `step`, which carries it out and returns whether it moved. The view takes
17/// these events over from the browser's own scrolling, setting `screen`'s
18/// `touch-action` to `none` so the browser doesn't pan or zoom with its
19/// touches first, and follows them for as long as the page lives, so call
20/// it once.
21///
22/// With [Ratzilla], the state is shared between drawing and scrolling, and
23/// the backend measures its cells:
24///
25/// ```ignore
26/// use std::{cell::RefCell, rc::Rc};
27///
28/// use ratmark::{MarkdownState, MarkdownView, Theme};
29/// use ratzilla::ratatui::Terminal;
30/// use ratzilla::{CellSized, DomBackend, WebRenderer};
31///
32/// let state = Rc::new(RefCell::new(MarkdownState::new(markdown)));
33/// let terminal = Terminal::new(DomBackend::new_by_element(screen.clone())?)?;
34///
35/// let (width, height) = terminal.backend().cell_size_css_px();
36/// let cell = (f64::from(width), f64::from(height));
37/// let scrolled = state.clone();
38/// ratmark::gesture::web::listen(&screen, cell, move |step| {
39///     step.apply(&mut scrolled.borrow_mut())
40/// })?;
41///
42/// let theme = Theme::default();
43/// terminal.draw_web(move |frame| {
44///     let view = MarkdownView::new(&theme);
45///     frame.render_stateful_widget(view, frame.area(), &mut state.borrow_mut());
46/// });
47/// ```
48///
49/// [Ratzilla]: https://github.com/orhun/ratzilla
50pub fn listen(
51    screen: &Element,
52    cell: (f64, f64),
53    step: impl FnMut(Step) -> bool + 'static,
54) -> Result<(), JsValue> {
55    if let Some(screen) = screen.dyn_ref::<HtmlElement>() {
56        screen.style().set_property("touch-action", "none")?;
57    }
58    let listener = Rc::new(RefCell::new(Listener {
59        gestures: Gestures::new(cell.0, cell.1),
60        step: Box::new(step),
61        screen: screen.clone(),
62        line: cell.1,
63        gliding: false,
64    }));
65    listen_wheel(&listener)?;
66    listen_touches(&listener)
67}
68
69type Shared = Rc<RefCell<Listener>>;
70
71struct Listener {
72    gestures: Gestures,
73    step: Box<dyn FnMut(Step) -> bool>,
74    screen: Element,
75    /// How tall a line is, in pixels, for wheels that go by lines.
76    line: f64,
77    /// Whether an animation frame is waiting to keep a flick going.
78    gliding: bool,
79}
80
81impl Listener {
82    /// Where `x`, `y` pixels in the window are in the view.
83    fn in_view(&self, x: i32, y: i32) -> (f64, f64) {
84        let rect = self.screen.get_bounding_client_rect();
85        (f64::from(x) - rect.left(), f64::from(y) - rect.top())
86    }
87
88    /// How far a wheel event goes, sideways and down, in pixels.
89    fn wheel_pixels(&self, e: &WheelEvent) -> (f64, f64) {
90        let pixels = |delta: f64| match e.delta_mode() {
91            WheelEvent::DOM_DELTA_LINE => delta * self.line,
92            WheelEvent::DOM_DELTA_PAGE => delta * f64::from(self.screen.client_height()),
93            _ => delta,
94        };
95        (pixels(e.delta_x()), pixels(e.delta_y()))
96    }
97}

99fn listen_wheel(listener: &Shared) -> Result<(), JsValue> {
100    let shared = listener.clone();
101    let on_wheel = Closure::<dyn FnMut(WheelEvent)>::new(move |e: WheelEvent| {
102        e.prevent_default();
103        let listener = &mut *shared.borrow_mut();
104        let at = listener.in_view(e.client_x(), e.client_y());
105        let delta = listener.wheel_pixels(&e);
106        let step = &mut listener.step;
107        listener.gestures.wheel(at, delta, e.time_stamp(), step);
108    });
109    listener
110        .borrow()
111        .screen
112        .add_event_listener_with_callback_and_add_event_listener_options(
113            "wheel",
114            on_wheel.as_ref().unchecked_ref(),
115            &not_passive(),
116        )?;
117    on_wheel.forget();
118    Ok(())
119}

121/// A finger drags the page along, and a flick keeps it gliding.
122fn listen_touches(listener: &Shared) -> Result<(), JsValue> {
123    let shared = listener.clone();
124    let on_move = Closure::<dyn FnMut(TouchEvent)>::new(move |e: TouchEvent| {
125        e.prevent_default();
126        let listener = &mut *shared.borrow_mut();
127        let Some(at) = first_touch(listener, &e) else {
128            return;
129        };
130        let step = &mut listener.step;
131        listener.gestures.touch_move(at, e.time_stamp(), step);
132    })
133    .into_js_value();
134
135    let shared = listener.clone();
136    let on_end = Closure::<dyn FnMut(TouchEvent)>::new(move |_: TouchEvent| {
137        let mut listener = shared.borrow_mut();
138        if listener.gestures.touch_end() && !listener.gliding {
139            listener.gliding = true;
140            drop(listener);
141            glide(shared.clone());
142        }
143    })
144    .into_js_value();
145
146    // A touch's events all go to the element it started on, even once a
147    // redraw has taken that element out of the page, when they no longer
148    // bubble up to the view. So the rest of the touch is followed on the
149    // element itself, the innermost one, which may be inside a shadow root.
150    let shared = listener.clone();
151    let on_start = Closure::<dyn FnMut(TouchEvent)>::new(move |e: TouchEvent| {
152        {
153            let listener = &mut *shared.borrow_mut();
154            if let Some(at) = first_touch(listener, &e) {
155                listener.gestures.touch_start(at, e.time_stamp());
156            }
157        }
158        let Ok(target) = e.composed_path().get(0).dyn_into::<EventTarget>() else {
159            return;
160        };
161        // Adding the same listener to an element again does nothing.
162        let _ = target.add_event_listener_with_callback_and_add_event_listener_options(
163            "touchmove",
164            on_move.unchecked_ref(),
165            &not_passive(),
166        );
167        for kind in ["touchend", "touchcancel"] {
168            let _ = target.add_event_listener_with_callback(kind, on_end.unchecked_ref());
169        }
170    });
171    listener
172        .borrow()
173        .screen
174        .add_event_listener_with_callback("touchstart", on_start.as_ref().unchecked_ref())?;
175    on_start.forget();
176    Ok(())
177}

179/// Keeps a flick going a frame at a time, until it stops.
180fn glide(shared: Shared) {
181    let Some(window) = web_sys::window() else {
182        shared.borrow_mut().gliding = false;
183        return;
184    };
185    let looping = shared.clone();
186    let frame = Closure::once_into_js(move |now: f64| {
187        let more = {
188            let listener = &mut *looping.borrow_mut();
189            let step = &mut listener.step;
190            let more = listener.gestures.glide(now, step);
191            listener.gliding = more;
192            more
193        };
194        if more {
195            glide(looping);
196        }
197    });
198    if window
199        .request_animation_frame(frame.unchecked_ref())
200        .is_err()
201    {
202        shared.borrow_mut().gliding = false;
203    }
204}

206/// Where the first finger of a touch is in the view, in pixels.
207fn first_touch(listener: &Listener, e: &TouchEvent) -> Option<(f64, f64)> {
208    let touch = e.touches().get(0)?;
209    Some(listener.in_view(touch.client_x(), touch.client_y()))
210}

212/// Listener options that let the view take the event over from the
213/// browser's own scrolling.
214fn not_passive() -> AddEventListenerOptions {
215    let options = AddEventListenerOptions::new();
216    options.set_passive(false);
217    options
