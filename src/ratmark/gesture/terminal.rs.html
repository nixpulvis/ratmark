1//! Following a terminal's mouse wheel with [`Gestures`].
2//!
3//! A terminal reports the wheel a tick at a time, up, down, left or right,
4//! and a trackpad swipe that's a little diagonal sends ticks both ways. A
5//! [`Wheel`] keeps each gesture going one way, so the swipe either scrolls
6//! or shifts the wide code under the pointer, not both.
7//!
8//! It counts time with std's [`Instant`], so it isn't built for
9//! `wasm32-unknown-unknown`, which has no clock for it. In a browser, use
10//! the `web` feature's `web::listen` instead.
11
12use std::time::Instant;
13
14use super::{Gestures, Step};
15
16/// A terminal's mouse wheel, moving the view a few cells a tick, one way a
17/// gesture.
18///
19/// The terminal's events are the host's to read, so each gives a `match`
20/// from its wheel events to ticks, like this one for crossterm's:
21///
22/// ```
23/// use ratmark::MarkdownState;
24/// use ratmark::gesture::terminal::Wheel;
25/// use ratatui_crossterm::crossterm::event::{MouseEvent, MouseEventKind};
26/// # use ratatui_crossterm::crossterm::event::KeyModifiers;
27/// # use ratatui_core::{buffer::Buffer, layout::Rect, widgets::StatefulWidget};
28///
29/// /// How far a crossterm wheel event goes, right and down, in ticks.
30/// fn ticks(kind: MouseEventKind) -> Option<(isize, isize)> {
31///     match kind {
32///         MouseEventKind::ScrollDown => Some((0, 1)),
33///         MouseEventKind::ScrollUp => Some((0, -1)),
34///         MouseEventKind::ScrollRight => Some((1, 0)),
35///         MouseEventKind::ScrollLeft => Some((-1, 0)),
36///         _ => None,
37///     }
38/// }
39///
40/// let markdown = (1..=20).map(|n| format!("{n}\n\n")).collect::<String>();
41/// let mut state = MarkdownState::new(markdown);
42/// # let area = Rect::new(0, 0, 10, 5);
43/// # ratmark::MarkdownView::new(&Default::default())
44/// #     .render(area, &mut Buffer::empty(area), &mut state);
45/// // Three lines, or columns, a tick.
46/// let mut wheel = Wheel::new(3);
47///
48/// // On each mouse event.
49/// # let mouse = MouseEvent {
50/// #     kind: MouseEventKind::ScrollDown,
51/// #     column: 4,
52/// #     row: 2,
53/// #     modifiers: KeyModifiers::NONE,
54/// # };
55/// if let Some(ticks) = ticks(mouse.kind) {
56///     wheel.tick(mouse.column, mouse.row, ticks, |step| step.apply(&mut state));
57/// }
58/// assert_eq!(state.scroll(), 3);
59/// ```
60#[derive(Debug, Clone)]
61pub struct Wheel {
62    gestures: Gestures,
63    /// What the gestures' times count from.
64    start: Instant,
65    /// How many lines, or columns, a tick moves.
66    cells: f64,
67}
68
69impl Wheel {
70    /// A wheel moving `cells` lines, or columns, a tick.
71    pub fn new(cells: u16) -> Self {
72        Self {
73            // Ticks and the pointer are in whole cells already.
74            gestures: Gestures::new(1.0, 1.0),
75            start: Instant::now(),
76            cells: f64::from(cells),
77        }
78    }
79
80    /// Ticks over `column`, `row`, going `ticks.0` right and `ticks.1` down,
81    /// handing each [`Step`] to `step`, which carries it out and returns
82    /// whether it moved. Each gesture goes the way its first tick does,
83    /// until a pause, or until it runs into the end.
84    pub fn tick(
85        &mut self,
86        column: u16,
87        row: u16,
88        ticks: (isize, isize),
89        step: impl FnOnce(Step) -> bool,
90    ) {
91        let time = self.start.elapsed().as_secs_f64() * 1000.0;
92        self.tick_at(column, row, ticks, time, step);
93    }
94
95    fn tick_at(
96        &mut self,
97        column: u16,
98        row: u16,
99        (x, y): (isize, isize),
100        time: f64,
101        step: impl FnOnce(Step) -> bool,
102    ) {
103        let at = (f64::from(column), f64::from(row));
104        let delta = (x as f64 * self.cells, y as f64 * self.cells);
105        self.gestures.wheel(at, delta, time, step);
106    }
107}

109#[cfg(test)]
110mod tests {
111    use super::*;
112    use crate::gesture::Axis;
113
114    #[test]
115    fn ticks_move_one_way_over_the_pointer() {
116        let mut wheel = Wheel::new(3);
117        let mut steps = Vec::new();
118        let mut record = |step: Step| {
119            steps.push(step);
120            true
121        };
122        wheel.tick_at(7, 4, (1, 0), 0.0, &mut record);
123        // A diagonal swipe's tick the other way, in the same gesture.
124        wheel.tick_at(7, 4, (0, 1), 20.0, &mut record);
125        wheel.tick_at(7, 4, (-1, 0), 40.0, &mut record);
126        // After a pause, a new gesture can go the other way.
127        wheel.tick_at(7, 4, (0, 1), 500.0, &mut record);
128        let step = |axis, cells| Step {
129            axis,
130            cells,
131            column: 7,
132            row: 4,
133        };
134        assert_eq!(
135            steps,
136            [
137                step(Axis::Horizontal, 3),
138                step(Axis::Horizontal, -3),
139                step(Axis::Vertical, 3),
140            ]
141        );
142    }
