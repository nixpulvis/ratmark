1//! Code blocks too wide for the page, drawn shifted sideways.
2
3use std::ops::Range;
4
5use ratatui_core::style::Style;
6use ratatui_core::text::{Line, Span};
7
8use crate::wrap::Piece;
9
10/// A code block too wide for the page. Its lines are cut off at the page's
11/// edge, with a `›` where there's more, and can be drawn shifted sideways to
12/// show the rest.
13///
14/// ```
15/// use ratmark::{Rendered, Theme};
16///
17/// let rendered = Rendered::render("```\nlet wide = 1;\n```\n", 10, &Theme::plain());
18/// let code = &rendered.overflows[0];
19/// assert_eq!(code.max_shift(), 5);
20///
21/// // Laid out cut off at the edge.
22/// assert_eq!(rendered.lines[code.lines.start].to_string(), " let wide›");
23/// // Drawn shifted to its end, over the same columns.
24/// assert_eq!(code.row(0, 5).to_string(), "‹ide = 1; ");
25/// ```
26#[derive(Debug, Clone, PartialEq)]
27pub struct Overflow {
28    /// Which of the document's code blocks this is, counting from 0, the
29    /// same at any width.
30    pub block: usize,
31    /// The lines the block is drawn on.
32    pub lines: Range<usize>,
33    /// The column the block starts at, including its padding, if any.
34    pub x: u16,
35    /// How many columns the code gets, not counting the padding.
36    pub width: u16,
37    /// The code, line by line, uncut.
38    pub(crate) code: Vec<Vec<Piece>>,
39    pub(crate) style: Style,
40    /// Whether there's a column of padding on either side, from
41    /// [`Theme::code_block_padding`].
42    pub(crate) padded: bool,
43}
44
45impl Overflow {
46    /// How far the block can be shifted before its widest line runs out.
47    pub fn max_shift(&self) -> usize {
48        let widest = self
49            .code
50            .iter()
51            .map(|line| width_of(line))
52            .max()
53            .unwrap_or(0);
54        widest.saturating_sub(usize::from(self.width))
55    }
56
57    /// The block's `i`th line with its padding, shifted `shift` columns to
58    /// the left, `‹` and `›` showing where code is out of sight.
59    pub fn row(&self, i: usize, shift: usize) -> Line<'static> {
60        let spans = self.pieces(i, shift);
61        Line::from_iter(spans.into_iter().map(|p| Span::styled(p.text, p.style)))
62    }
63
64    /// The columns the block is drawn across, padding included.
65    pub(crate) fn columns(&self) -> Range<u16> {
66        self.x..self.x + self.width + if self.padded { 2 } else { 0 }
67    }
68
69    /// The pieces of [`Overflow::row`].
70    pub(crate) fn pieces(&self, i: usize, shift: usize) -> Vec<Piece> {
71        let shift = shift.min(self.max_shift());
72        let width = usize::from(self.width);
73        let code = &self.code[i];
74        let code_width = width_of(code);
75        let after = code_width > shift + width;
76        // Unpadded, the marks take the code's first and last columns, so
77        // leave the code at least one.
78        let before = shift > 0 && code_width > 0 && (self.padded || width > usize::from(after));
79        let (start, end) = if self.padded {
80            (shift, shift + width)
81        } else {
82            (
83                shift + usize::from(before),
84                shift + width - usize::from(after),
85            )
86        };
87        let marker =
88            |more: bool, c: &'static str| Piece::new(if more { c } else { " " }, self.style);
89        let mut pieces = Vec::new();
90        if self.padded || before {
91            pieces.push(marker(before, "‹"));
92        }
93        let mut x = 0;
94        for piece in code {
95            let from = x;
96            x += piece.width();
97            if x <= start || from >= end {
98                continue;
99            }
100            let shown = piece.cut(start.saturating_sub(from)..end - from);
101            if !shown.text.is_empty() {
102                pieces.push(Piece::new(shown.text, self.style.patch(piece.style)));
103            }
104        }
105        let used = x.clamp(start, end) - start;
106        pieces.push(Piece::new(" ".repeat(end - start - used), self.style));
107        if self.padded || after {
108            pieces.push(marker(after, "›"));
109        }
110        pieces
111    }
112}
113
114/// How wide a line of pieces is.
115fn width_of(line: &[Piece]) -> usize {
116    line.iter().map(Piece::width).sum()
117}
