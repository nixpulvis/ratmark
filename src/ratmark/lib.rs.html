1//! Render Markdown as a [Ratatui] [`StatefulWidget`], or into Ratatui text.
2//!
3//! Draw a [`MarkdownView`] in any Ratatui [`Terminal`], then drive its state
4//! from input:
5//!
6//! ```
7//! use ratmark::{MarkdownState, MarkdownView, Theme};
8//! # let backend = ratatui_core::backend::TestBackend::new(40, 5);
9//! # let mut terminal = ratatui_core::terminal::Terminal::new(backend)?;
10//!
11//! let theme = Theme::default();
12//! let mut state = MarkdownState::new("# Hello\n\nRead [the docs](https://docs.rs/ratmark).");
13//!
14//! // `terminal` is your app's `ratatui_core::terminal::Terminal`, like
15//! // `Terminal::new(CrosstermBackend::new(io::stdout()))?` from ratatui-crossterm.
16//! terminal.draw(|frame| {
17//!     let view = MarkdownView::new(&theme);
18//!     frame.render_stateful_widget(view, frame.area(), &mut state);
19//! })?;
20//!
21//! // On input, scroll, focus the next link, or find the link under a click.
22//! state.scroll_by(1);
23//! state.focus_next();
24//! let focused = state.focused().map(|link| link.url.as_str()); // Some("https://docs.rs/ratmark")
25//! let clicked = state.link_at(5, 2).map(|link| link.url.as_str()); // Some("https://docs.rs/ratmark")
26//! # assert_eq!(focused, Some("https://docs.rs/ratmark"));
27//! # assert_eq!(clicked, Some("https://docs.rs/ratmark"));
28//! # Ok::<(), Box<dyn std::error::Error>>(())
29//! ```
30//!
31//! [`MarkdownState`] holds the document, its layout and where you are in it.
32//! [`MarkdownView`] wraps the document to the width it's given, rewrapping
33//! when the width changes, and draws it scrolled with the focused link
34//! highlighted.
35//!
36//! Underneath, [`Rendered::render`] lays a document out for a given width
37//! and returns its lines along with where each link landed.
38//!
39//! ```
40//! use ratmark::{Rendered, Theme};
41//!
42//! let theme = Theme::default();
43//! let markdown = "# Hi\n\nSee [the docs](https://docs.rs).";
44//! let rendered = Rendered::render(markdown, 40, &theme);
45//!
46//! assert_eq!(rendered.lines[0].to_string(), "Hi");
47//! assert_eq!(rendered.lines[0].spans[0].style, theme.headings[0]);
48//! assert_eq!(rendered.links[0].url, "https://docs.rs");
49//! ```
50//!
51//! A [`Theme`] holds every style. [`Theme::ansi_dark`] is the default, and
52//! [`Theme::ansi_light`] suits a light background. Both use
53//! terminal color numbers, so the terminal's palette decides what they look
54//! like. [`Theme::plain`] uses no colors at all. Every style is a public
55//! field, so making your own is a struct literal.
56//!
57//! CommonMark plus tables, footnotes, strikethrough, task lists, definition
58//! lists and math are supported. Code too wide for the page scrolls
59//! sideways rather than wrapping.
60//!

156/// A rendered document.
157#[derive(Debug, Clone, Default, PartialEq)]
158pub struct Rendered {
159    /// The document's lines, already wrapped. Draw them without wrapping.
160    pub lines: Vec<Line<'static>>,
161    /// Every link, in the order they appear.
162    pub links: Vec<Link>,
163    /// Every heading, in order, for jumping to a section or listing them.
164    pub headings: Vec<Heading>,
165    /// Code blocks too wide for the page, which [`Rendered::lines`] shows
166    /// cut off.
167    pub overflows: Vec<Overflow>,
168}
169
170impl Rendered {
171    /// Lays `markdown` out in lines at most `width` columns wide.
172    ///
173    /// Front matter is laid out like any other text.
174    #[cfg_attr(
175        feature = "_front-matter",
176        doc = "Split it off first with [`front_matter::FrontMatter::split`] to leave it out."
177    )]
178    pub fn render(markdown: &str, width: u16, theme: &Theme) -> Self {
179        render::Renderer::new(theme, width).run(markdown)
180    }
181
182    /// The lines as a [`Text`].
183    pub fn text(&self) -> Text<'static> {
184        Text::from(self.lines.clone())
185    }
186
187    /// The link drawn at column `x` of line `line`, if there is one.
188    pub fn link_at(&self, line: usize, x: u16) -> Option<&Link> {
189        self.links.iter().find(|link| {
190            link.regions
191                .iter()
192                .any(|r| r.line == line && (r.start..r.end).contains(&x))
193        })
194    }
195}

197/// A heading and the line it starts on.
198#[derive(Debug, Clone, PartialEq, Eq)]
199pub struct Heading {
200    /// 1 for `#` through 6 for `######`.
201    pub level: u8,
202    /// Its text, without markup.
203    pub text: String,
204    /// The line it starts on.
205    pub line: usize,
206}
207
208/// A link and the places it was drawn.
209#[derive(Debug, Clone, PartialEq, Eq)]
210pub struct Link {
211    /// Where it goes, as written in the document.
212    pub url: String,
213    /// Where the link's text landed. A link that wraps has one region per
214    /// line.
215    pub regions: Vec<Region>,
216}
217
218/// A stretch of one line, in columns from the start of the line.
219#[derive(Debug, Clone, Copy, PartialEq, Eq)]
220pub struct Region {
221    /// The line it's on.
222    pub line: usize,
223    /// The first column it covers.
224    pub start: u16,
225    /// The column after the last it covers.
226    pub end: u16,
227}
