3use ratatui_core::style::{Color, Modifier, Style};
4
5/// The styles used for each kind of Markdown element.
6///
7/// Inline styles like [`Theme::strong`] are patched over the style of the
8/// block they appear in, so a theme only needs to set what changes.
9#[derive(Debug, Clone, PartialEq)]
10pub struct Theme {
11    /// Plain text, which everything else is patched over.
12    pub text: Style,
13    /// Heading text, by level from `#` to `######`.
14    pub headings: [Style; 6],
15    /// The style of `#` marks drawn in front of each heading, Markdown or
16    /// HTML, to show its level. `None`, the default, draws no marks.
17    pub heading_marker: Option<Style>,
18    /// `*Emphasis*`.
19    pub emphasis: Style,
20    /// `**Strong**` text.
21    pub strong: Style,
22    /// `~~Struck through~~` text.
23    pub strikethrough: Style,
24    /// Link text.
25    pub link: Style,
26    /// An image's `[image: alt]` label, patched over the link style.
27    pub image: Style,
28    /// Inline `code`.
29    pub code: Style,
30    /// Fenced and indented code. The background fills the block's width.
31    pub code_block: Style,
32    /// Leave a column on either side of each code block, where `‹` and `›`
33    /// show code cut off at the edges. Without it the code takes the full
34    /// width, and the marks are drawn over its first and last columns.
35    pub code_block_padding: bool,
36    /// Text in a block quote.
37    pub quote: Style,
38    /// The bar down the side of a block quote.
39    pub quote_marker: Style,
40    /// List bullets and numbers, and task list boxes.
41    pub list_marker: Style,
42    /// A horizontal rule.
43    pub rule: Style,
44    /// Inline and display math, shown as written.
45    pub math: Style,
46    /// Footnote references and labels, like `[1]`.
47    pub footnote: Style,
48    /// The lines between a table's columns and rows.
49    pub table_border: Style,
50    /// A table's header row.
51    pub table_header: Style,
52    /// Draw a rule between body rows too, not just under the header.
53    pub table_row_rules: bool,
54    /// Text from `<small>` tags, used for things like dates under a title.
55    pub small: Style,
56    /// The syntect theme used to color code blocks, when the `syntax`
57    /// feature is on. Only its foreground colors are used. Besides syntect's
58    /// bundled themes there's `"ansi"`, which colors code with terminal
59    /// color numbers, like [`Theme::ansi_dark`].
60    ///
61    /// As in [bat], a syntect color with an alpha of `00` names a terminal
62    /// color by the number in its red, and an alpha of `01` keeps the code
63    /// block's own foreground.
64    ///
65    /// [bat]: https://github.com/sharkdp/bat
66    pub syntax: Option<&'static str>,
67    /// Styles for elements with these classes, written like CSS class
68    /// selectors, `.note` or `.note.warning`, and patched in order over
69    /// everything inside a matching element. A code block inside takes its
70    /// background. Classes come from HTML, with the `html` feature, and from
71    /// a kramdown attribute list right after inline code, like
72    /// `` `x`{:.new} ``, with the `kramdown` feature.
73    pub classes: Vec<(String, Style)>,
74}
75
76impl Default for Theme {
77    fn default() -> Self {
78        Self::ansi_dark()
79    }
80}

82impl Theme {
83    /// Colors picked from the terminal's palette, for a dark background.
84    ///
85    /// Each color is a terminal color number, like 4 for blue, never a fixed
86    /// RGB value, so the terminal's palette decides what it looks like. Text
87    /// and the background keep the terminal's defaults. Code, inline or in
88    /// blocks, sits on gray 234 from the 256-color ramp, which suits a dark
89    /// background.
90    pub fn ansi_dark() -> Self {
91        Self::ansi(234)
92    }
93
94    /// Colors picked from the terminal's palette, for a light background. The
95    /// same as [`Theme::ansi_dark`], but code sits on the light gray 255.
96    pub fn ansi_light() -> Self {
97        Self::ansi(255)
98    }
99
100    /// Colors from the terminal's palette, with gray `shade` behind all code.
101    fn ansi(shade: u8) -> Self {
102        const GREEN: Color = Color::Indexed(2);
103        const BLUE: Color = Color::Indexed(4);
104        const MAGENTA: Color = Color::Indexed(5);
105        const CYAN: Color = Color::Indexed(6);
106        // Bright black, the terminal's muted gray.
107        const GRAY: Color = Color::Indexed(8);
108
109        let bold = Style::new().add_modifier(Modifier::BOLD);
110        let gray = Style::new().fg(GRAY);
111        Self {
112            text: Style::new(),
113            headings: [
114                bold.add_modifier(Modifier::UNDERLINED),
115                bold,
116                bold,
117                bold,
118                bold,
119                bold,
120            ],
121            heading_marker: None,
122            emphasis: Style::new().add_modifier(Modifier::ITALIC),
123            strong: bold,
124            strikethrough: Style::new().add_modifier(Modifier::CROSSED_OUT),
125            link: Style::new().fg(BLUE).add_modifier(Modifier::UNDERLINED),
126            image: Style::new().fg(MAGENTA),
127            code: Style::new().fg(GREEN).bg(Color::Indexed(shade)),
128            code_block: Style::new().bg(Color::Indexed(shade)),
129            code_block_padding: true,
130            quote: Style::new().add_modifier(Modifier::ITALIC),
131            quote_marker: gray,
132            list_marker: Style::new(),
133            rule: gray,
134            math: Style::new().fg(CYAN),
135            footnote: gray,
136            table_border: gray,
137            table_header: bold,
138            table_row_rules: true,
139            small: gray,
140            syntax: Some("ansi"),
141            classes: Vec::new(),
142        }
143    }
144
145    /// No colors at all, only bold, italic, underline, dim and reversed text.
146    /// It reads the same on any background, and in terminals without color.
147    pub fn plain() -> Self {
148        let bold = Style::new().add_modifier(Modifier::BOLD);
149        let dim = Style::new().add_modifier(Modifier::DIM);
150        Self {
151            text: Style::new(),
152            headings: [
153                bold.add_modifier(Modifier::UNDERLINED),
154                bold,
155                bold,
156                bold,
157                bold,
158                bold,
159            ],
160            heading_marker: None,
161            emphasis: Style::new().add_modifier(Modifier::ITALIC),
162            strong: bold,
163            strikethrough: Style::new().add_modifier(Modifier::CROSSED_OUT),
164            link: Style::new().add_modifier(Modifier::UNDERLINED),
165            image: Style::new().add_modifier(Modifier::UNDERLINED),
166            code: Style::new().add_modifier(Modifier::REVERSED),
167            code_block: Style::new(),
168            code_block_padding: true,
169            quote: Style::new().add_modifier(Modifier::ITALIC),
170            quote_marker: dim,
171            list_marker: Style::new(),
172            rule: dim,
173            math: Style::new().add_modifier(Modifier::ITALIC),
174            footnote: Style::new(),
175            table_border: dim,
176            table_header: bold,
177            table_row_rules: true,
178            small: dim,
179            syntax: None,
180            classes: Vec::new(),
181        }
182    }

184    /// The style for an element with `classes`, from [`Theme::classes`],
185    /// or `None` when no selector matches.
186    #[cfg(any(feature = "html", feature = "kramdown"))]
187    pub(crate) fn class_style<'c>(
188        &self,
189        classes: impl IntoIterator<Item = &'c str> + Clone,
190    ) -> Option<Style> {
191        let has = |class: &str| classes.clone().into_iter().any(|c| c == class);
192        self.classes
193            .iter()
194            .filter(|(selector, _)| {
195                let mut wanted = selector.split('.').filter(|c| !c.is_empty()).peekable();
196                wanted.peek().is_some() && wanted.all(has)
197            })
198            .map(|(_, style)| *style)
199            .reduce(|style, more| style.patch(more))
200    }
