1//! Scrolling a terminal UI with a wheel, a trackpad or a finger, a cell at a
2//! time.
3//!
4//! Each gesture only goes one way, up and down, or sideways, so a swipe
5//! that's a little diagonal doesn't do both. Hosts that report movement in
6//! pixels, like a browser, move the view as far as the hand does, a line or
7//! a column at a time once one's worth adds up, and a flick keeps the view
8//! gliding for a bit, slowing down, like a page in the browser does.
9//!
10//! [`Gestures`] does the arithmetic, from pixels and timestamps, and hands
11//! each whole cell of movement to a callback as a [`Step`]. The callback
12//! moves whatever the app scrolls, and returns whether it moved, which ends
13//! the gesture at the end of what there is to scroll. Each host feeds it its
14//! own events:
15//!
16//! - In a terminal, [`terminal::Wheel`] takes the wheel's ticks, like
17//!   crossterm's `MouseEventKind::ScrollDown`.
18//! - In a browser, the `web` feature's `web::listen` follows the wheel and
19//!   touch events over the element a view is drawn in, like [Ratzilla]'s.
20//! - Anywhere else that reports movement in pixels, call [`Gestures`]'
21//!   methods from its events, and [`Gestures::glide`] once a frame after
22//!   [`Gestures::touch_end`] says a touch was a flick.
23//!
24//! Nothing here draws, or depends on what does. A [`Step`] is only so many
25//! cells one way, over the cell the gesture is on.
26//!
27//! ```
28//! use ratjest::{Axis, Gestures, Step};
29//!
30//! // Twenty lines in a view five tall, so the top line is 0 through 15.
31//! let mut top: isize = 0;
32//! let mut scroll = |step: Step| {
33//!     if step.axis != Axis::Vertical {
34//!         return false;
35//!     }
36//!     let before = top;
37//!     top = (top + step.cells).clamp(0, 15);
38//!     top != before
39//! };
40//!
41//! // Cells 8 pixels wide and 16 tall.
42//! let mut gestures = Gestures::new(8.0, 16.0);
43//!
44//! // Half a line doesn't move yet, the rest of a line and a half does.
45//! gestures.wheel((0.0, 0.0), (0.0, 8.0), 0.0, &mut scroll);
46//! gestures.wheel((0.0, 0.0), (0.0, 24.0), 10.0, &mut scroll);
47//! assert_eq!(top, 2);
48//! ```
49//!
50//! [Ratzilla]: https://github.com/ratatui/ratzilla

59/// Slowest a flick glides, in pixels a millisecond.
60const MIN_GLIDE: f64 = 0.05;
61
62/// How far a finger goes before its touch picks a way to move, in pixels.
63const LOCK_DISTANCE: f64 = 6.0;
64
65/// How long between wheel events before the next starts a new gesture, in
66/// milliseconds.
67const WHEEL_GESTURE: f64 = 200.0;
68
69/// How long a glide's first frame is taken to be, in milliseconds.
70const FRAME: f64 = 16.0;
71
72/// The way a gesture moves the view. A gesture only ever moves one way.
73#[derive(Debug, Clone, Copy, PartialEq, Eq)]
74pub enum Axis {
75    /// Up and down.
76    Vertical,
77    /// Left and right.
78    Horizontal,
79}
80
81impl Axis {
82    /// The way a move of `x`, `y` pixels mostly goes.
83    fn of(x: f64, y: f64) -> Self {
84        if x.abs() > y.abs() {
85            Axis::Horizontal
86        } else {
87            Axis::Vertical
88        }
89    }
90
91    /// The part of a move of `x`, `y` pixels that goes this way.
92    fn part(self, x: f64, y: f64) -> f64 {
93        match self {
94            Axis::Vertical => y,
95            Axis::Horizontal => x,
96        }
97    }
98}

100/// A gesture's move of whole cells.
101#[derive(Debug, Clone, Copy, PartialEq, Eq)]
102pub struct Step {
103    /// Which way it goes.
104    pub axis: Axis,
105    /// How many lines down, or columns right, it goes. Negative goes up, or
106    /// left.
107    pub cells: isize,
108    /// The column of the cell the gesture is over: the pointer for a wheel,
109    /// or where the finger came down for a touch.
110    pub column: u16,
111    /// The row of the cell the gesture is over.
112    pub row: u16,
113}
114
115/// Turns wheel and touch movement, in pixels, into [`Step`]s of whole
116/// cells, keeping the part of a cell left over for the gesture's next
117/// event.
118///
119/// Positions are in pixels from the view's top left corner, and times in
120/// milliseconds from any fixed point, like a browser event's `timeStamp`.
121/// Each method that moves takes a `step` callback, which carries out a
122/// [`Step`] and returns whether it moved, which it doesn't at the end.
123#[derive(Debug, Clone)]
124pub struct Gestures {
125    /// How wide and tall the view's cells are, in pixels.
126    cell: (f64, f64),
127    /// The part of a cell left over, down and sideways.
128    carried: (f64, f64),
129    /// Where the pointer, or the finger, last was.
130    pointer: (f64, f64),
131    /// The way the wheel gesture goes, and when its last event came, while
132    /// it hasn't run into the end.
133    wheel: Option<(Axis, f64)>,
134    touch: Touch,
135}
136
137impl Gestures {
138    /// Gestures over a view with cells `width` by `height` pixels.
139    pub fn new(width: f64, height: f64) -> Self {
140        Self {
141            cell: (width, height),
142            carried: (0.0, 0.0),
143            pointer: (0.0, 0.0),
144            wheel: None,
145            touch: Touch::default(),
146        }
147    }
148
149    /// A wheel event at `at`, going `delta` pixels right and down, at
150    /// `time`. A wheel gesture, like a swipe on a trackpad, goes the way its
151    /// first event mostly goes, until a pause, or until it runs into the
152    /// end, like a sideways flick's momentum at the end of a line.
153    pub fn wheel(
154        &mut self,
155        at: (f64, f64),
156        delta: (f64, f64),
157        time: f64,
158        step: impl FnOnce(Step) -> bool,
159    ) {
160        self.pointer = at;
161        let (x, y) = delta;
162        let axis = match self.wheel {
163            Some((axis, last)) if time - last < WHEEL_GESTURE => axis,
164            _ => {
165                let axis = Axis::of(x, y);
166                self.start(axis);
167                axis
168            }
169        };
170        let moved = self.move_by(axis, axis.part(x, y), step);
171        self.wheel = moved.then_some((axis, time));
172    }
173
174    /// A finger coming down at `at`, at `time`, catching any glide.
175    pub fn touch_start(&mut self, at: (f64, f64), time: f64) {
176        self.pointer = at;
177        self.touch = Touch {
178            start: Some(at),
179            last: Some(at),
180            at: time,
181            ..Touch::default()
182        };
183    }
184
185    /// The finger moving to `at`, at `time`, dragging the view along. It
186    /// doesn't move until it's gone far enough to pick which way it goes.
187    pub fn touch_move(&mut self, at: (f64, f64), time: f64, step: impl FnOnce(Step) -> bool) {
188        let touch = &mut self.touch;
189        let (Some(start), Some(last)) = (touch.start, touch.last) else {
190            return;
191        };
192        let axis = match touch.axis {
193            Some(axis) => axis,
194            None => {
195                let (x, y) = (at.0 - start.0, at.1 - start.1);
196                if x.abs().max(y.abs()) < LOCK_DISTANCE {
197                    return;
198                }
199                let axis = *touch.axis.insert(Axis::of(x, y));
200                self.start(axis);
201                axis
202            }
203        };
204        let touch = &mut self.touch;
205        touch.last = Some(at);
206        let pixels = axis.part(last.0 - at.0, last.1 - at.1);
207        let ms = (time - touch.at).max(1.0);
208        // Smoothed, so one jittery event doesn't decide a flick.
209        touch.speed = 0.8 * pixels / ms + 0.2 * touch.speed;
210        touch.at = time;
211        let _ = self.move_by(axis, pixels, step);
212    }
213
214    /// The finger lifting. Returns whether it was a flick, which glides on
215    /// with [`glide`](Self::glide) once a frame.
216    pub fn touch_end(&mut self) -> bool {
217        let touch = &mut self.touch;
218        touch.start = None;
219        touch.last = None;
220        touch.gliding = touch.axis.take().filter(|_| touch.speed.abs() > MIN_GLIDE);
221        touch.glided = None;
222        touch.gliding.is_some()
223    }
224
225    /// Keeps a flick going, for a frame drawn at `time`. Returns whether
226    /// it's still gliding, until it slows to a stop, runs into the end, or
227    /// another touch catches it.
228    pub fn glide(&mut self, time: f64, step: impl FnOnce(Step) -> bool) -> bool {
229        let touch = &mut self.touch;
230        let Some(axis) = touch.gliding else {
231            return false;
232        };
233        let ms = touch
234            .glided
235            .map_or(FRAME, |last| (time - last).clamp(1.0, 100.0));
236        touch.glided = Some(time);
237        let pixels = touch.speed * ms;
238        // About 5% slower every 16 milliseconds.
239        touch.speed *= 0.95_f64.powf(ms / FRAME);
240        if touch.speed.abs() <= MIN_GLIDE {
241            touch.gliding = None;
242        }
243        if !self.move_by(axis, pixels, step) {
244            self.touch.gliding = None;
245        }
246        self.touch.gliding.is_some()
247    }
248
249    /// Starts a gesture going `axis`, dropping what the last one left over,
250    /// so this one doesn't start part of a cell in.
251    fn start(&mut self, axis: Axis) {
252        *self.carried_mut(axis) = 0.0;
253    }
254
255    /// Moves `pixels` further along `axis`, and says whether the view
256    /// moved, which it doesn't at the end.
257    fn move_by(&mut self, axis: Axis, pixels: f64, step: impl FnOnce(Step) -> bool) -> bool {
258        let size = axis.part(self.cell.0, self.cell.1);
259        let carried = self.carried_mut(axis);
260        let total = *carried + pixels;
261        let cells = (total / size).trunc();
262        *carried = total - cells * size;
263        if cells == 0.0 {
264            return true;
265        }
266        let at = |pixels: f64, size: f64| (pixels / size).max(0.0) as u16;
267        step(Step {
268            axis,
269            cells: cells as isize,
270            column: at(self.pointer.0, self.cell.0),
271            row: at(self.pointer.1, self.cell.1),
272        })
273    }
274
275    fn carried_mut(&mut self, axis: Axis) -> &mut f64 {
276        match axis {
277            Axis::Vertical => &mut self.carried.1,
278            Axis::Horizontal => &mut self.carried.0,
279        }
280    }
281}

283/// A finger on the screen, and how fast it was going.
284#[derive(Debug, Clone, Default)]
285struct Touch {
286    /// Where the finger came down, and where it was last.
287    start: Option<(f64, f64)>,
288    last: Option<(f64, f64)>,
289    /// The way this touch moves the view, once it's gone far enough to tell.
290    axis: Option<Axis>,
291    /// When the last touch event came.
292    at: f64,
293    /// In pixels a millisecond, down or right when positive.
294    speed: f64,
295    /// The way a flick is gliding, and when its last frame was.
296    gliding: Option<Axis>,
297    glided: Option<f64>,
298}

304    /// A callback recording the steps it takes, which move until there are
305    /// `end` of them.
306    fn record(steps: &mut Vec<Step>, end: usize) -> impl FnOnce(Step) -> bool + '_ {
307        move |step| {
308            steps.push(step);
309            steps.len() < end
310        }
311    }
312
313    fn cells(steps: &[Step]) -> Vec<(Axis, isize)> {
314        steps.iter().map(|s| (s.axis, s.cells)).collect()
315    }

317    #[test]
318    fn moves_a_cell_at_a_time() {
319        let mut gestures = Gestures::new(10.0, 10.0);
320        let mut steps = Vec::new();
321        for (pixels, time) in [(4.0, 0.0), (4.0, 10.0), (25.0, 20.0), (-30.0, 30.0)] {
322            gestures.wheel((0.0, 0.0), (0.0, pixels), time, record(&mut steps, 10));
323        }
324        // 8 pixels isn't a cell yet, 33 is 3 with 3 over, and -27 is -2.
325        assert_eq!(cells(&steps), [(Axis::Vertical, 3), (Axis::Vertical, -2)]);
326        // After a pause a new gesture starts from nothing, rather than the -7
327        // left over, so two 5s make a cell.
328        gestures.wheel((0.0, 0.0), (0.0, 5.0), 500.0, record(&mut steps, 10));
329        gestures.wheel((0.0, 0.0), (0.0, 5.0), 510.0, record(&mut steps, 10));
330        assert_eq!(steps.len(), 3);
331    }

333    #[test]
334    fn wheels_move_one_way_until_the_end() {
335        let mut gestures = Gestures::new(10.0, 10.0);
336        let mut steps = Vec::new();
337        gestures.wheel((25.0, 35.0), (20.0, 5.0), 0.0, record(&mut steps, 10));
338        // Still sideways, though now it's mostly down.
339        gestures.wheel((25.0, 35.0), (10.0, 40.0), 50.0, record(&mut steps, 10));
340        assert_eq!(
341            cells(&steps),
342            [(Axis::Horizontal, 2), (Axis::Horizontal, 1)]
343        );
344        // Over the cell at the pointer.
345        assert_eq!((steps[0].column, steps[0].row), (2, 3));
346        // Running into the end frees the next event to go the other way.
347        gestures.wheel((0.0, 0.0), (10.0, 0.0), 60.0, record(&mut steps, 3));
348        gestures.wheel((0.0, 0.0), (0.0, 10.0), 70.0, record(&mut steps, 10));
349        assert_eq!(steps[3].axis, Axis::Vertical);
350    }

352    #[test]
353    fn touches_move_one_way() {
354        let mut gestures = Gestures::new(1.0, 1.0);
355        let mut steps = Vec::new();
356        gestures.touch_start((100.0, 100.0), 0.0);
357        // Too short a move to tell which way.
358        gestures.touch_move((103.0, 102.0), 10.0, record(&mut steps, 10));
359        assert!(steps.is_empty());
360        gestures.touch_move((90.0, 104.0), 20.0, record(&mut steps, 10));
361        // Still sideways, though now it's mostly down.
362        gestures.touch_move((88.0, 140.0), 30.0, record(&mut steps, 10));
363        assert_eq!(
364            cells(&steps),
365            [(Axis::Horizontal, 10), (Axis::Horizontal, 2)]
366        );
367        // Over where the finger came down.
368        assert_eq!((steps[1].column, steps[1].row), (100, 100));
369        // Held still before lifting, so it doesn't glide.
370        gestures.touch_move((88.0, 140.0), 500.0, record(&mut steps, 10));
371        assert!(!gestures.touch_end());
372    }

374    /// A finger flicking `gestures` up the view, and lifting.
375    fn flick(gestures: &mut Gestures) -> bool {
376        gestures.touch_start((100.0, 300.0), 0.0);
377        for (i, y) in [280.0, 240.0, 200.0].into_iter().enumerate() {
378            gestures.touch_move((100.0, y), 10.0 * (i + 1) as f64, |_| true);
379        }
380        gestures.touch_end()
381    }

383    #[test]
384    fn flicks_glide_until_they_stop() {
385        let mut gestures = Gestures::new(1.0, 1.0);
386        let mut steps = Vec::new();
387        assert!(flick(&mut gestures));
388        let mut time = 100.0;
389        while gestures.glide(time, record(&mut steps, usize::MAX)) {
390            time += 16.0;
391        }
392        // It glides on down, then slows to a stop.
393        assert!(steps.len() > 1);
394        assert!(
395            steps
396                .iter()
397                .all(|s| s.axis == Axis::Vertical && s.cells > 0)
398        );
399        assert!(steps[0].cells > steps[steps.len() - 1].cells);
400        assert!(!gestures.glide(time + 16.0, record(&mut steps, usize::MAX)));
401
402        // It stops at the end.
403        assert!(flick(&mut gestures));
404        assert!(!gestures.glide(100.0, |_| false));
405
406        // Or when another touch catches it.
407        assert!(flick(&mut gestures));
408        gestures.touch_start((100.0, 300.0), 200.0);
409        assert!(!gestures.glide(216.0, |_| panic!("caught glides on")));
410    }
