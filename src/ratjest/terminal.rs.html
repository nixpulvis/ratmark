1//! Following a terminal's mouse wheel with [`Gestures`].
2//!
3//! A terminal reports the wheel a tick at a time, up, down, left or right,
4//! and a trackpad swipe that's a little diagonal sends ticks both ways. A
5//! [`Wheel`] keeps each gesture going one way, so the swipe goes either up
6//! and down or sideways, not both.
7//!
8//! It counts time with std's [`Instant`], so it isn't built for
9//! `wasm32-unknown-unknown`, which has no clock for it. In a browser, use
10//! the `web` feature's `web::listen` instead.
11
12use std::time::Instant;
13
14use ratatui_core::layout::Rect;
15
16use crate::{Gestures, Scroll};
17
18/// A terminal's mouse wheel, moving the view a few cells a tick, one way a
19/// gesture.
20///
21/// The terminal's events are the host's to read, so each gives a `match`
22/// from its wheel events to ticks, like this one for crossterm's:
23///
24/// ```
25/// use crossterm::event::{MouseEvent, MouseEventKind};
26/// # use crossterm::event::KeyModifiers;
27/// use ratatui::layout::Rect;
28/// use ratatui::widgets::ListState;
29/// use ratjest::Scroll;
30/// use ratjest::terminal::Wheel;
31///
32/// /// How far a crossterm wheel event goes, right and down, in ticks.
33/// fn ticks(kind: MouseEventKind) -> Option<(isize, isize)> {
34///     match kind {
35///         MouseEventKind::ScrollDown => Some((0, 1)),
36///         MouseEventKind::ScrollUp => Some((0, -1)),
37///         MouseEventKind::ScrollRight => Some((1, 0)),
38///         MouseEventKind::ScrollLeft => Some((-1, 0)),
39///         _ => None,
40///     }
41/// }
42///
43/// // A list, and where its items are drawn.
44/// let mut state = ListState::default();
45/// let area = Rect::new(0, 0, 30, 5);
46/// // Three rows, or columns, a tick.
47/// let mut wheel = Wheel::new(3);
48///
49/// // On each mouse event.
50/// # let mouse = MouseEvent {
51/// #     kind: MouseEventKind::ScrollDown,
52/// #     column: 4,
53/// #     row: 2,
54/// #     modifiers: KeyModifiers::NONE,
55/// # };
56/// if let Some(ticks) = ticks(mouse.kind) {
57///     wheel.tick(mouse.column, mouse.row, ticks, &mut state, area);
58/// }
59/// assert_eq!(state.offset(), 3);
60/// ```
61#[derive(Debug, Clone)]
62pub struct Wheel {
63    gestures: Gestures,
64    /// What the gestures' times count from.
65    start: Instant,
66    /// How many lines, or columns, a tick moves.
67    cells: f64,
68}
69
70impl Wheel {
71    /// A wheel moving `cells` lines, or columns, a tick.
72    pub fn new(cells: u16) -> Self {
73        Self {
74            // Ticks and the pointer are in whole cells already.
75            gestures: Gestures::new(1.0, 1.0),
76            start: Instant::now(),
77            cells: f64::from(cells),
78        }
79    }
80
81    /// Ticks over `column`, `row`, going `ticks.0` right and `ticks.1` down,
82    /// scrolling `scroll`, which is drawn in `area`. Each gesture goes the
83    /// way its first tick does, until a pause, or until it runs into the
84    /// end.
85    pub fn tick(
86        &mut self,
87        column: u16,
88        row: u16,
89        ticks: (isize, isize),
90        scroll: impl Scroll,
91        area: Rect,
92    ) {
93        let time = self.start.elapsed().as_secs_f64() * 1000.0;
94        self.tick_at(column, row, ticks, time, scroll, area);
95    }
96
97    fn tick_at(
98        &mut self,
99        column: u16,
100        row: u16,
101        (x, y): (isize, isize),
102        time: f64,
103        scroll: impl Scroll,
104        area: Rect,
105    ) {
106        let at = (f64::from(column), f64::from(row));
107        let delta = (x as f64 * self.cells, y as f64 * self.cells);
108        self.gestures.wheel(at, delta, time, scroll, area);
109    }
110}

112#[cfg(test)]
113mod tests {
114    use super::*;
115    use crate::{Axis, Step};
116
117    /// A view recording the steps it's given.
118    struct Record(Vec<Step>);
119
120    impl Scroll for Record {
121        fn scroll(&mut self, step: Step, _: Rect) -> bool {
122            self.0.push(step);
123            true
124        }
125    }
126
127    #[test]
128    fn ticks_move_one_way_over_the_pointer() {
129        let mut wheel = Wheel::new(3);
130        let area = Rect::new(0, 0, 80, 24);
131        let mut record = Record(Vec::new());
132        wheel.tick_at(7, 4, (1, 0), 0.0, &mut record, area);
133        // A diagonal swipe's tick the other way, in the same gesture.
134        wheel.tick_at(7, 4, (0, 1), 20.0, &mut record, area);
135        wheel.tick_at(7, 4, (-1, 0), 40.0, &mut record, area);
136        // After a pause, a new gesture can go the other way.
137        wheel.tick_at(7, 4, (0, 1), 500.0, &mut record, area);
138        let step = |axis, cells| Step {
139            axis,
140            cells,
141            column: 7,
142            row: 4,
143        };
144        assert_eq!(
145            record.0,
146            [
147                step(Axis::Horizontal, 3),
148                step(Axis::Horizontal, -3),
149                step(Axis::Vertical, 3),
150            ]
151        );
152    }
