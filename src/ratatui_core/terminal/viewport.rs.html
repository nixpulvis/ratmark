5/// The area of the terminal that Ratatui draws into.
6///
7/// A [`Viewport`] controls where widgets render and what [`Frame::area`] returns.
8///
9/// For a higher-level overview of viewports in the context of an application (including
10/// examples), see [`Terminal`].
11///
12/// Choose a viewport based on how the Ratatui UI should fit into the terminal:
13///
14/// - [`Viewport::Fullscreen`] for the standard case: your app owns the whole terminal surface.
15/// - [`Viewport::Inline`] when the UI should live inside a larger CLI flow, with normal terminal
16///   output above it.
17/// - [`Viewport::Fixed`] when Ratatui should render into one region of a terminal layout managed
18///   elsewhere.
19///
20/// In fullscreen mode, the viewport starts at (0, 0). In inline and fixed mode, the viewport may
21/// have a non-zero `x`/`y` origin; prefer using `Frame::area()` as your root layout rectangle.
22/// Code that assumes `(0, 0)` as the origin is therefore only correct for fullscreen viewports.
23///
24/// See [`Terminal::with_options`] for how to select a viewport, and [`Terminal::resize`] /
25/// [`Terminal::autoresize`] for resize behavior.
26///
27/// # Example
28///
29/// ```rust,no_run
30/// # #![allow(unexpected_cfgs)]
31/// # #[cfg(feature = "crossterm")]
32/// # {
33/// use ratatui::backend::CrosstermBackend;
34/// use ratatui::layout::{Constraint, Layout, Rect};
35/// use ratatui::{Terminal, TerminalOptions, Viewport};
36///
37/// let mut terminal = Terminal::with_options(
38///     CrosstermBackend::new(std::io::stdout()),
39///     TerminalOptions {
40///         viewport: Viewport::Fixed(Rect::new(10, 5, 20, 4)),
41///     },
42/// )?;
43///
44/// terminal.draw(|frame| {
45///     // `frame.area()` is `Rect::new(10, 5, 20, 4)`, not `(0, 0, 20, 4)`.
46///     let [title, body] =
47///         Layout::vertical([Constraint::Length(1), Constraint::Min(0)]).areas(frame.area());
48///
49///     frame.render_widget("panel title", title);
50///     frame.render_widget("render the body relative to the fixed viewport", body);
51/// })?;
52/// # }
53/// # Ok::<(), Box<dyn std::error::Error>>(())
54/// ```
55///
56/// [`Frame::area`]: crate::terminal::Frame::area
57/// [`Terminal`]: crate::terminal::Terminal
58/// [`Terminal::with_options`]: crate::terminal::Terminal::with_options
59/// [`Terminal::resize`]: crate::terminal::Terminal::resize
60/// [`Terminal::autoresize`]: crate::terminal::Terminal::autoresize
61#[derive(Debug, Default, Clone, Eq, PartialEq, Hash)]
62pub enum Viewport {
63    /// Draw into the entire terminal.
64    ///
65    /// This is the default viewport used by [`Terminal::new`].
66    ///
67    /// Choose this when the Ratatui app should own the whole terminal window.
68    ///
69    /// When the terminal size changes, Ratatui automatically resizes internal buffers during
70    /// [`Terminal::draw`] / [`Terminal::try_draw`].
71    ///
72    /// `Frame::area()` always starts at (0, 0).
73    ///
74    /// [`Terminal::new`]: crate::terminal::Terminal::new
75    /// [`Terminal::draw`]: crate::terminal::Terminal::draw
76    /// [`Terminal::try_draw`]: crate::terminal::Terminal::try_draw
77    #[default]
78    Fullscreen,
79    /// Draw the application inline with the rest of the terminal output.
80    ///
81    /// Choose this when the UI should appear inside a larger command-line flow instead of taking
82    /// over the entire terminal.
83    ///
84    /// The viewport spans the full terminal width and its top-left corner is anchored to column 0
85    /// of the current cursor row when the terminal is created and whenever it is recomputed during
86    /// resize. Ratatui reserves space for the requested height; if the cursor is near the bottom
87    /// of the screen, this may scroll the terminal so the viewport remains fully visible.
88    ///
89    /// The height is specified in rows and is clamped to the current terminal height.
90    ///
91    /// Inline viewports always span the full terminal width.
92    ///
93    /// For the full inline rendering model, including output inserted above the UI, see the
94    /// "Inline Viewport" section on [`Terminal`](crate::terminal::Terminal) and
95    /// [`Terminal::insert_before`](crate::terminal::Terminal::insert_before).
96    ///
97    /// [`Terminal::draw`]: crate::terminal::Terminal::draw
98    /// [`Terminal::try_draw`]: crate::terminal::Terminal::try_draw
99    Inline(u16),
100    /// Draw into a fixed region of the terminal.
101    ///
102    /// Choose this when Ratatui is responsible for only part of the screen, for example a panel in
103    /// a larger terminal layout managed by another renderer or by surrounding application code.
104    ///
105    /// Fixed viewports are not automatically resized. If the region should change (for example, on
106    /// terminal resize), call [`Terminal::resize`] yourself.
107    ///
108    /// The area is specified as a [`Rect`] in terminal coordinates.
109    ///
110    /// `Frame::area()` returns this rectangle as-is (including its `x`/`y` offset).
111    /// Ratatui does not keep this rectangle synchronized with backend resizes unless you call
112    /// [`Terminal::resize`] yourself.
113    ///
114    /// See also [`Terminal::with_options`] for initialization behavior.
115    ///
116    /// [`Terminal::with_options`]: crate::terminal::Terminal::with_options
117    /// [`Terminal::resize`]: crate::terminal::Terminal::resize
118    Fixed(Rect),
119}
120
121impl fmt::Display for Viewport {
122    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
123        match self {
124            Self::Fullscreen => write!(f, "Fullscreen"),
125            Self::Inline(height) => write!(f, "Inline({height})"),
126            Self::Fixed(area) => write!(f, "Fixed({area})"),
127        }
128    }
129}

137    #[test]
138    fn viewport_to_string() {
139        assert_eq!(Viewport::Fullscreen.to_string(), "Fullscreen");
140        assert_eq!(Viewport::Inline(5).to_string(), "Inline(5)");
141        assert_eq!(
142            Viewport::Fixed(Rect::new(0, 0, 5, 5)).to_string(),
143            "Fixed(5x5+0+0)"
