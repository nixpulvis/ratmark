1//! HTML rendering that matches kramdown's output, attributes included.
2//!
3//! Mirrors [`pulldown_cmark::html`][cmark]: the same functions, but taking
4//! the `(Event, Attributes)` pairs from
5//! [`Document::events`](crate::Document::events) instead of bare events.
6//!
7//! [cmark]: https://docs.rs/pulldown-cmark/0.13/pulldown_cmark/html/index.html
8//!
9//! ```
10//! use pulldown_kram::{Document, Extensions, html};
11//!
12//! let document = Document::new("# Hi\n\nSee *this*{:.new}.\n", Extensions::all());
13//! let mut out = String::new();
14//! html::push_html(&mut out, document.events());
15//! assert_eq!(out, "<h1 id=\"hi\">Hi</h1>\n<p>See <em class=\"new\">this</em>.</p>\n");
16//! ```
17//!
18//! As in kramdown, footnotes are numbered in the order they're first
19//! referenced and rendered at the end, with links back to each reference.
20//! Footnote definitions are held until then; everything else is written as
21//! it arrives. Math alone in its paragraph is written as `\[…\]`, and inline
22//! math as `\(…\)`, for MathJax to render. Code isn't highlighted: a code
23//! block's language becomes a `language-…` class on its `<code>`. Empty
24//! table cells contain a no-break space. Raw HTML in the document is written
25//! unchanged.
26//!
27//! kramdown's converter options, like `footnote_nr`, aren't supported, nor
28//! are the ids kramdown gives the links in tables of contents. References to
29//! undefined footnotes can only occur with pulldown-cmark's
30//! `ENABLE_OLD_FOOTNOTES`, and link nowhere.

42/// An event, with its attributes.
43type Attributed<'a> = (Event<'a>, Attributes<'a>);
44
45/// Renders the events of `iter` as HTML, appending to `s`.
46pub fn push_html<'a, I>(s: &mut String, iter: I)
47where
48    I: Iterator<Item = (Event<'a>, Attributes<'a>)>,
49{
50    write_html_fmt(s, iter).expect("writing to a String to succeed");
51}
52
53/// Renders the events of `iter` as HTML to `writer`, returning its error if
54/// a write fails.
55pub fn write_html_fmt<'a, I, W>(writer: W, iter: I) -> fmt::Result
56where
57    I: Iterator<Item = (Event<'a>, Attributes<'a>)>,
58    W: fmt::Write,
59{
60    let mut html = HtmlWriter {
61        out: writer,
62        footnotes: Footnotes::default(),
63        table: CurrentTable::default(),
64    };
65    html.write(Lookahead::new(iter))?;
66    html.write_footnotes()
67}

69/// Renders the events of `iter` as HTML to `writer`, returning the first
70/// error it produces. Writes aren't buffered, so wrapping `writer` in a
71/// [`BufWriter`](std::io::BufWriter) avoids many small writes.
72pub fn write_html_io<'a, I, W>(writer: W, iter: I) -> io::Result<()>
73where
74    I: Iterator<Item = (Event<'a>, Attributes<'a>)>,
75    W: io::Write,
76{
77    let mut writer = IoWriter {
78        inner: writer,
79        error: None,
80    };
81    write_html_fmt(&mut writer, iter).map_err(|fmt::Error| {
82        writer
83            .error
84            .take()
85            .unwrap_or_else(|| io::Error::other(fmt::Error))
86    })
87}

89/// An `io::Write` as a `fmt::Write`, keeping the error it gave.
90struct IoWriter<W> {
91    inner: W,
92    error: Option<io::Error>,
93}
94
95impl<W: io::Write> fmt::Write for IoWriter<W> {
96    fn write_str(&mut self, s: &str) -> fmt::Result {
97        self.inner.write_all(s.as_bytes()).map_err(|error| {
98            self.error = Some(error);
99            fmt::Error
100        })
101    }
102}

104/// Events, with the next few in view.
105struct Lookahead<'a, I> {
106    iter: I,
107    ahead: VecDeque<Attributed<'a>>,
108}
109
110impl<'a, I: Iterator<Item = Attributed<'a>>> Lookahead<'a, I> {
111    fn new(iter: I) -> Self {
112        Self {
113            iter,
114            ahead: VecDeque::new(),
115        }
116    }
117
118    fn next(&mut self) -> Option<Attributed<'a>> {
119        self.ahead.pop_front().or_else(|| self.iter.next())
120    }
121
122    /// The event `n` after the next one, `0` being the next.
123    fn peek(&mut self, n: usize) -> Option<&Event<'a>> {
124        while self.ahead.len() <= n {
125            self.ahead.push_back(self.iter.next()?);
126        }
127        Some(&self.ahead[n].0)
128    }
129
130    /// The rest of the element whose start was just taken, up to its end,
131    /// which is taken but left out.
132    fn element(&mut self) -> Vec<Attributed<'a>> {
133        let mut content = Vec::new();
134        let mut depth = 1usize;
135        while let Some(item) = self.next() {
136            match item.0 {
137                Event::Start(_) => depth += 1,
138                Event::End(_) => {
139                    depth -= 1;
140                    if depth == 0 {
141                        break;
142                    }
143                }
144                _ => {}
145            }
146            content.push(item);
147        }
148        content
149    }
150
151    /// If the paragraph just started holds only math, that math, taken
152    /// along with the paragraph's end.
153    fn math_alone(&mut self) -> Option<CowStr<'a>> {
154        let alone = matches!(self.peek(0), Some(Event::DisplayMath(_)))
155            && matches!(self.peek(1), Some(Event::End(TagEnd::Paragraph)));
156        if !alone {
157            return None;
158        }
159        let (Event::DisplayMath(math), _) = self.next()? else {
160            return None;
161        };
162        self.next();
163        Some(math)
164    }
165}

167/// Footnotes, numbered in the order they're first referred to, and what
168/// they hold, kept to be written after the rest.
169#[derive(Default)]
170struct Footnotes<'a> {
171    /// Those referred to, in the order of their first references.
172    referred: Vec<Referred<'a>>,
173    /// The index in `referred` of each, by its label in lowercase, as
174    /// pulldown-cmark matches labels ignoring case.
175    indices: HashMap<String, usize>,
176    /// What each footnote defined but not yet written holds, by its label
177    /// in lowercase.
178    definitions: HashMap<String, Vec<Attributed<'a>>>,
179    /// How many of those referred to have been looked at for writing.
180    looked_at: usize,
181}
182
183/// A footnote to write.
184struct Footnote<'a> {
185    /// Its index among those referred to.
186    index: usize,
187    /// Its label, as first referred to.
188    label: CowStr<'a>,
189    /// What it holds.
190    content: Vec<Attributed<'a>>,
191}
192
193/// A footnote referred to.
194struct Referred<'a> {
195    /// Its label, as first referred to.
196    label: CowStr<'a>,
197    /// How many times it's been referred to.
198    references: usize,
199}
200
201/// A reference to a footnote.
202struct Reference<'a> {
203    /// The footnote's label, as first referred to.
204    label: CowStr<'a>,
205    /// The footnote's number, from 1.
206    number: usize,
207    /// Which reference to the footnote this is, from 0.
208    nth: usize,
209}
210
211impl<'a> Footnotes<'a> {
212    /// Keeps what the footnote `label` holds, unless it's been defined
213    /// already, as the first definition wins.
214    fn define(&mut self, label: &str, content: Vec<Attributed<'a>>) {
215        self.definitions
216            .entry(label.to_lowercase())
217            .or_insert(content);
218    }
219
220    /// Counts a reference to the footnote `label`, numbering it if it's the
221    /// first.
222    fn refer(&mut self, label: CowStr<'a>) -> Reference<'a> {
223        let referred = &mut self.referred;
224        let index = *self.indices.entry(label.to_lowercase()).or_insert_with(|| {
225            referred.push(Referred {
226                label,
227                references: 0,
228            });
229            referred.len() - 1
230        });
231        let referred = &mut self.referred[index];
232        referred.references += 1;
233        Reference {
234            label: referred.label.clone(),
235            number: index + 1,
236            nth: referred.references - 1,
237        }
238    }
239
240    /// The next footnote referred to that's defined, to write. Footnotes
241    /// first referred to while it's written come after it.
242    fn next_to_write(&mut self) -> Option<Footnote<'a>> {
243        while let Some(referred) = self.referred.get(self.looked_at) {
244            let index = self.looked_at;
245            self.looked_at += 1;
246            if let Some(content) = self.definitions.remove(&referred.label.to_lowercase()) {
247                return Some(Footnote {
248                    index,
249                    label: referred.label.clone(),
250                    content,
251                });
252            }
253        }
254        None
255    }
256
257    /// How many times the footnote at `index` has been referred to.
258    fn references(&self, index: usize) -> usize {
259        self.referred[index].references
260    }
261}

263/// The table being written.
264#[derive(Default)]
265struct CurrentTable {
266    /// The alignment of its columns.
267    alignments: Vec<Alignment>,
268    /// The column of the next cell in the row.
269    column: usize,
270    /// Whether its body has started, after its header row.
271    in_body: bool,
272}
273
274impl CurrentTable {
275    /// Starts a row, returning whether it starts the body.
276    fn start_row(&mut self) -> bool {
277        self.column = 0;
278        !std::mem::replace(&mut self.in_body, true)
279    }
280
281    /// Starts the header row.
282    fn start_header(&mut self) {
283        self.column = 0;
284    }
285
286    /// The name of a cell in the current row.
287    fn cell(&self) -> &'static str {
288        if self.in_body { "td" } else { "th" }
289    }
290
291    /// The style aligning the next cell, if it's aligned, moving on to the
292    /// cell after it.
293    fn next_alignment(&mut self) -> Option<&'static str> {
294        let style = match self.alignments.get(self.column) {
295            Some(Alignment::Left) => Some("text-align: left"),
296            Some(Alignment::Center) => Some("text-align: center"),
297            Some(Alignment::Right) => Some("text-align: right"),
298            Some(Alignment::None) | None => None,
299        };
300        self.column += 1;
301        style
302    }
303}

305/// Writes events as HTML to `out`.
306struct HtmlWriter<'a, W> {
307    out: W,
308    footnotes: Footnotes<'a>,
309    table: CurrentTable,
310}
311
312impl<'a, W: fmt::Write> HtmlWriter<'a, W> {
313    /// Writes `events`, keeping footnote definitions for the end.
314    fn write<I: Iterator<Item = Attributed<'a>>>(
315        &mut self,
316        mut events: Lookahead<'a, I>,
317    ) -> fmt::Result {
318        while let Some((event, attributes)) = events.next() {
319            match event {
320                Event::Start(Tag::FootnoteDefinition(label)) => {
321                    self.footnotes.define(&label, events.element());
322                }
323                Event::Start(Tag::Paragraph) => match events.math_alone() {
324                    Some(math) => self.math_block(&math, &attributes)?,
325                    None => self.open("p", &attributes, &[])?,
326                },
327                Event::Start(Tag::Image {
328                    dest_url, title, ..
329                }) => self.image(&dest_url, &title, &attributes, events.element())?,
330                Event::Start(Tag::CodeBlock(kind)) => {
331                    self.code_block(&kind, attributes, events.element())?;
332                }
333                // Metadata isn't content.
334                Event::Start(Tag::MetadataBlock(_)) => {
335                    events.element();
336                }
337                // kramdown writes an empty cell with a no-break space.
338                Event::Start(Tag::TableCell)
339                    if matches!(events.peek(0), Some(Event::End(TagEnd::TableCell))) =>
340                {
341                    self.event(Event::Start(Tag::TableCell), &attributes)?;
342                    self.out.write_char('\u{a0}')?;
343                }
344                event => self.event(event, &attributes)?,
345            }
346        }
347        Ok(())
348    }
349
350    /// Writes `event`, with its `attributes`.
351    fn event(&mut self, event: Event<'a>, attributes: &Attributes) -> fmt::Result {
352        match event {
353            Event::Start(tag) => self.start(tag, attributes),
354            Event::End(tag) => self.end(tag),
355            Event::Text(text) => self.escaped(&text),
356            Event::Code(code) => {
357                self.open("code", attributes, &[])?;
358                self.escaped(&code)?;
359                self.out.write_str("</code>")
360            }
361            Event::InlineMath(math) => {
362                self.out.write_str("\\(")?;
363                self.escaped(&math)?;
364                self.out.write_str("\\)")
365            }
366            Event::DisplayMath(math) => {
367                self.out.write_str("\\[")?;
368                self.escaped(&math)?;
369                self.out.write_str("\\]")
370            }
371            Event::Html(html) | Event::InlineHtml(html) => self.out.write_str(&html),
372            Event::FootnoteReference(label) => self.footnote_reference(label),
373            Event::SoftBreak => self.out.write_str("\n"),
374            Event::HardBreak => self.out.write_str("<br />\n"),
375            Event::Rule => {
376                self.empty("hr", attributes, &[])?;
377                self.out.write_str("\n")
378            }
379            Event::TaskListMarker(checked) => {
380                let mut pairs = vec![
381                    ("type", "checkbox"),
382                    ("class", "task-list-item-checkbox"),
383                    ("disabled", "disabled"),
384                ];
385                if checked {
386                    pairs.push(("checked", "checked"));
387                }
388                self.empty("input", &Attributes::default(), &pairs)
389            }
390        }
391    }
392
393    /// Writes the start of `tag`, with its `attributes`.
394    fn start(&mut self, tag: Tag<'a>, attributes: &Attributes) -> fmt::Result {
395        let name = match tag {
396            Tag::Paragraph => "p",
397            Tag::BlockQuote(_) => "blockquote",
398            Tag::List(None) => "ul",
399            Tag::List(Some(1)) => "ol",
400            Tag::List(Some(start)) => {
401                return self.open("ol", attributes, &[("start", &start.to_string())]);
402            }
403            Tag::Item => "li",
404            Tag::DefinitionList => "dl",
405            Tag::DefinitionListTitle => "dt",
406            Tag::DefinitionListDefinition => "dd",
407            Tag::Emphasis => "em",
408            Tag::Strong => "strong",
409            Tag::Strikethrough => "del",
410            Tag::Superscript => "sup",
411            Tag::Subscript => "sub",
412            Tag::Heading {
413                level,
414                id,
415                classes,
416                attrs,
417            } => return self.heading(level, attributes, id, classes, attrs),
418            Tag::Link {
419                link_type,
420                dest_url,
421                title,
422                ..
423            } => return self.link(link_type, &dest_url, &title, attributes),
424            Tag::Table(alignments) => {
425                self.table = CurrentTable {
426                    alignments,
427                    ..CurrentTable::default()
428                };
429                "table"
430            }
431            Tag::TableHead => {
432                self.table.start_header();
433                self.out.write_str("<thead>")?;
434                "tr"
435            }
436            Tag::TableRow => {
437                if self.table.start_row() {
438                    self.out.write_str("<tbody>")?;
439                }
440                "tr"
441            }
442            Tag::TableCell => {
443                let cell = self.table.cell();
444                return match self.table.next_alignment() {
445                    Some(style) => self.open(cell, attributes, &[("style", style)]),
446                    None => self.open(cell, attributes, &[]),
447                };
448            }
449            // HTML is written as it comes, and the rest whole by `write`.
450            Tag::HtmlBlock
451            | Tag::CodeBlock(_)
452            | Tag::Image { .. }
453            | Tag::FootnoteDefinition(_)
454            | Tag::MetadataBlock(_) => return Ok(()),
455        };
456        self.open(name, attributes, &[])
457    }
458
459    /// Writes the end of `tag`.
460    fn end(&mut self, tag: TagEnd) -> fmt::Result {
461        self.out.write_str(match tag {
462            TagEnd::Paragraph => "</p>\n",
463            TagEnd::Heading(level) => return writeln!(self.out, "</{level}>"),
464            TagEnd::BlockQuote(_) => "</blockquote>\n",
465            TagEnd::List(false) => "</ul>\n",
466            TagEnd::List(true) => "</ol>\n",
467            TagEnd::Item => "</li>\n",
468            TagEnd::DefinitionList => "</dl>\n",
469            TagEnd::DefinitionListTitle => "</dt>\n",
470            TagEnd::DefinitionListDefinition => "</dd>\n",
471            TagEnd::Emphasis => "</em>",
472            TagEnd::Strong => "</strong>",
473            TagEnd::Strikethrough => "</del>",
474            TagEnd::Superscript => "</sup>",
475            TagEnd::Subscript => "</sub>",
476            TagEnd::Link => "</a>",
477            TagEnd::Table if self.table.in_body => "</tbody></table>\n",
478            TagEnd::Table => "</table>\n",
479            TagEnd::TableHead => "</tr></thead>\n",
480            TagEnd::TableRow => "</tr>\n",
481            TagEnd::TableCell if self.table.in_body => "</td>",
482            TagEnd::TableCell => "</th>",
483            TagEnd::HtmlBlock
484            | TagEnd::CodeBlock
485            | TagEnd::Image
486            | TagEnd::FootnoteDefinition
487            | TagEnd::MetadataBlock(_) => "",
488        })
489    }

491    /// Writes the start of a heading, with pulldown-cmark's own attributes
492    /// for it under those of attribute lists.
493    fn heading(
494        &mut self,
495        level: HeadingLevel,
496        attributes: &Attributes,
497        id: Option<CowStr<'a>>,
498        classes: Vec<CowStr<'a>>,
499        pairs: Vec<(CowStr<'a>, Option<CowStr<'a>>)>,
500    ) -> fmt::Result {
501        let mut all = Attributes {
502            id: attributes.id.clone().or(id),
503            classes,
504            pairs: pairs
505                .into_iter()
506                .map(|(key, value)| (key, value.unwrap_or_else(|| "".into())))
507                .collect(),
508        };
509        all.classes.extend(attributes.classes.iter().cloned());
510        all.pairs.extend(attributes.pairs.iter().cloned());
511        self.open(level, &all, &[])
512    }

514    /// Writes the start of a link, an email address's to `mailto:` it.
515    fn link(
516        &mut self,
517        link_type: LinkType,
518        dest_url: &str,
519        title: &str,
520        attributes: &Attributes,
521    ) -> fmt::Result {
522        let mailto;
523        let href = if link_type == LinkType::Email && !dest_url.starts_with("mailto:") {
524            mailto = format!("mailto:{dest_url}");
525            &mailto
526        } else {
527            dest_url
528        };
529        let mut pairs = vec![("href", href)];
530        if !title.is_empty() {
531            pairs.push(("title", title));
532        }
533        self.open("a", attributes, &pairs)
534    }

536    /// Writes an image, its `content` written as its alternative text.
537    fn image(
538        &mut self,
539        dest_url: &str,
540        title: &str,
541        attributes: &Attributes,
542        content: Vec<Attributed<'a>>,
543    ) -> fmt::Result {
544        let mut alt = String::new();
545        for (event, _) in content {
546            if let Event::Text(text) | Event::Code(text) | Event::InlineMath(text) = event {
547                alt.push_str(&text);
548            }
549        }
550        let mut pairs = vec![("src", dest_url), ("alt", &alt)];
551        if !title.is_empty() {
552            pairs.push(("title", title));
553        }
554        self.empty("img", attributes, &pairs)
555    }

557    /// Writes `math` as a block, in a div if it has attributes.
558    fn math_block(&mut self, math: &str, attributes: &Attributes) -> fmt::Result {
559        let in_div = !attributes.is_empty();
560        if in_div {
561            self.open("div", attributes, &[])?;
562        }
563        self.out.write_str("\\[")?;
564        self.escaped(math)?;
565        self.out.write_str("\\]\n")?;
566        if in_div {
567            self.out.write_str("</div>\n")?;
568        }
569        Ok(())
570    }

572    /// Writes a code block, as kramdown does without a highlighter: its
573    /// language is the first `language-` class, the fence's coming first,
574    /// written on the `code`, with the rest of the attributes on the `pre`.
575    /// The fence's language is without the highlighter's options after a
576    /// `?`, like `php?start_inline=1`. One newline ends the code, whatever
577    /// it ends with, and the class `show-whitespaces` marks its spaces and
578    /// tabs.
579    fn code_block(
580        &mut self,
581        kind: &CodeBlockKind,
582        mut attributes: Attributes<'a>,
583        content: Vec<Attributed<'a>>,
584    ) -> fmt::Result {
585        let mut code = String::new();
586        for (event, _) in content {
587            if let Event::Text(text) = event {
588                code.push_str(&text);
589            }
590        }
591        let code = code.strip_suffix('\n').unwrap_or(&code);
592        if let CodeBlockKind::Fenced(info) = kind
593            && let Some(language) = info.split_whitespace().next()
594        {
595            let language = language
596                .char_indices()
597                .skip(1)
598                .find(|&(_, c)| c == '?')
599                .map_or(language, |(i, _)| &language[..i]);
600            let class = format!("language-{language}");
601            if !attributes.has_class(&class) {
602                attributes.classes.insert(0, class.into());
603            }
604        }
605        let show_whitespaces = attributes.has_class("show-whitespaces");
606        let language = attributes
607            .classes
608            .iter()
609            .position(|class| class.starts_with("language-"))
610            .map(|i| attributes.classes.remove(i));
611
612        self.open("pre", &attributes, &[])?;
613        match &language {
614            Some(language) => self.open("code", &Attributes::default(), &[("class", language)])?,
615            None => self.out.write_str("<code>")?,
616        }
617        if show_whitespaces {
618            self.whitespaces(code)?;
619        } else {
620            self.escaped(code)?;
621        }
622        self.out.write_str("\n</code></pre>\n")
623    }

625    /// Writes a reference to the footnote `label`, numbered in the order of
626    /// first references, as kramdown does.
627    fn footnote_reference(&mut self, label: CowStr<'a>) -> fmt::Result {
628        let Reference { label, number, nth } = self.footnotes.refer(label);
629        self.out.write_str("<sup id=\"")?;
630        self.reference_id(&label, nth)?;
631        self.out.write_str("\"><a href=\"#fn:")?;
632        self.escaped(&label)?;
633        write!(
634            self.out,
635            "\" class=\"footnote\" rel=\"footnote\" role=\"doc-noteref\">{number}</a></sup>"
636        )
637    }

639    /// Writes the footnotes referred to, after the document, as kramdown
640    /// does.
641    fn write_footnotes(&mut self) -> fmt::Result {
642        if self.footnotes.referred.is_empty() {
643            return Ok(());
644        }
645        self.out
646            .write_str("<div class=\"footnotes\" role=\"doc-endnotes\">\n<ol>\n")?;
647        while let Some(footnote) = self.footnotes.next_to_write() {
648            self.footnote(footnote)?;
649        }
650        self.out.write_str("</ol>\n</div>\n")
651    }

653    /// Writes `footnote`, with links back to each reference to it in its
654    /// last paragraph, or in one of their own if it doesn't end with one.
655    fn footnote(&mut self, footnote: Footnote<'a>) -> fmt::Result {
656        let Footnote {
657            index,
658            label,
659            mut content,
660        } = footnote;
661        self.out.write_str("<li id=\"fn:")?;
662        self.escaped(&label)?;
663        self.out.write_str("\">\n")?;
664        let in_paragraph = matches!(content.last(), Some((Event::End(TagEnd::Paragraph), _)));
665        if in_paragraph {
666            content.pop();
667        }
668        self.write(Lookahead::new(content.into_iter()))?;
669        if !in_paragraph {
670            self.out.write_str("<p>")?;
671        }
672        for nth in 0..self.footnotes.references(index) {
673            if in_paragraph || nth > 0 {
674                self.out.write_char('\u{a0}')?;
675            }
676            self.out.write_str("<a href=\"#")?;
677            self.reference_id(&label, nth)?;
678            self.out
679                .write_str("\" class=\"reversefootnote\" role=\"doc-backlink\">\u{21a9}")?;
680            if nth > 0 {
681                write!(self.out, "<sup>{}</sup>", nth + 1)?;
682            }
683            self.out.write_str("</a>")?;
684        }
685        self.out.write_str("</p>\n</li>\n")
686    }
687
688    /// Writes the id of the `nth` reference to the footnote `label`, from 0:
689    /// `fnref:label` for the first, and `fnref:label:1` and on after.
690    fn reference_id(&mut self, label: &str, nth: usize) -> fmt::Result {
691        self.out.write_str("fnref:")?;
692        self.escaped(label)?;
693        if nth > 0 {
694            write!(self.out, ":{nth}")?;
695        }
696        Ok(())
697    }
698
699    /// Writes a start tag, `<name …>`.
700    fn open(
701        &mut self,
702        name: impl fmt::Display,
703        attributes: &Attributes,
704        pairs: &[(&str, &str)],
705    ) -> fmt::Result {
706        write!(self.out, "<{name}")?;
707        self.attributes(attributes, pairs)?;
708        self.out.write_str(">")
709    }
710
711    /// Writes an element with no content or end tag, `<name … />`.
712    fn empty(
713        &mut self,
714        name: &str,
715        attributes: &Attributes,
716        pairs: &[(&str, &str)],
717    ) -> fmt::Result {
718        write!(self.out, "<{name}")?;
719        self.attributes(attributes, pairs)?;
720        self.out.write_str(" />")
721    }
722
723    /// Writes `attributes`, with the element's own `pairs` before their
724    /// pairs, as HTML attributes. A blank id is left out, as kramdown leaves
725    /// it out.
726    fn attributes(&mut self, attributes: &Attributes, pairs: &[(&str, &str)]) -> fmt::Result {
727        if let Some(id) = attributes.id.as_ref().filter(|id| !id.trim().is_empty()) {
728            self.attribute("id", id)?;
729        }
730        if !attributes.classes.is_empty() {
731            self.out.write_str(" class=\"")?;
732            for (i, class) in attributes.classes.iter().enumerate() {
733                if i > 0 {
734                    self.out.write_str(" ")?;
735                }
736                self.escaped(class)?;
737            }
738            self.out.write_str("\"")?;
739        }
740        for &(key, value) in pairs {
741            self.attribute(key, value)?;
742        }
743        for (key, value) in &attributes.pairs {
744            self.attribute(key, value)?;
745        }
746        Ok(())
747    }
748
749    /// Writes ` key="value"`.
750    fn attribute(&mut self, key: &str, value: &str) -> fmt::Result {
751        write!(self.out, " {key}=\"")?;
752        self.escaped(value)?;
753        self.out.write_str("\"")
754    }
755
756    /// Writes `text` with HTML's special characters escaped.
757    fn escaped(&mut self, text: &str) -> fmt::Result {
758        let mut rest = text;
759        while let Some(at) = rest.find(['&', '<', '>', '"']) {
760            self.out.write_str(&rest[..at])?;
761            self.out.write_str(match rest.as_bytes()[at] {
762                b'&' => "&amp;",
763                b'<' => "&lt;",
764                b'>' => "&gt;",
765                _ => "&quot;",
766            })?;
767            rest = &rest[at + 1..];
768        }
769        self.out.write_str(rest)
770    }
771
772    /// Writes `code`, escaped, with each space and tab in a span saying
773    /// whether it starts a line, ends one or is in between, as kramdown
774    /// marks them.
775    fn whitespaces(&mut self, code: &str) -> fmt::Result {
776        for (n, line) in code.split('\n').enumerate() {
777            if n > 0 {
778                self.out.write_char('\n')?;
779            }
780            let mut rest = line;
781            let mut start = true;
782            while !rest.is_empty() {
783                let text = rest.find([' ', '\t']).unwrap_or(rest.len());
784                self.escaped(&rest[..text])?;
785                start &= text == 0;
786                rest = &rest[text..];
787                let run = rest.find(|c| c != ' ' && c != '\t').unwrap_or(rest.len());
788                let place = if start {
789                    "-l"
790                } else if run == rest.len() {
791                    "-r"
792                } else {
793                    ""
794                };
795                for c in rest[..run].chars() {
796                    match c {
797                        '\t' => write!(self.out, "<span class=\"ws-tab{place}\">\t</span>")?,
798                        _ => write!(self.out, "<span class=\"ws-space{place}\">&#8901;</span>")?,
799                    }
800                }
801                rest = &rest[run..];
802                start = false;
803            }
804        }
805        Ok(())
806    }
