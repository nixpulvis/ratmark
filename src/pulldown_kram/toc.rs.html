1//! kramdown's tables of contents: lists marked `{:toc}`, replaced with
2//! links to the document's headings. The tables share a budget of items,
3//! as [`Extensions::ENABLE_TABLES_OF_CONTENTS`] explains.
4//!
5//! [`Extensions::ENABLE_TABLES_OF_CONTENTS`]: crate::Extensions::ENABLE_TABLES_OF_CONTENTS
6
7use std::ops::Range;
8
9use pulldown_cmark::{CowStr, Event, LinkType, Tag, TagEnd};
10
11use crate::read::Item;

13/// A document's headings as a tree, for its tables of contents. Each
14/// heading sits under the nearest heading before it with a higher level.
15pub(crate) struct Toc<'a> {
16    entries: Vec<Entry<'a>>,
17    /// The entries under each entry.
18    children: Vec<Vec<usize>>,
19    /// The entries under none.
20    roots: Vec<usize>,
21}
22
23/// A heading listed in a table of contents.
24struct Entry<'a> {
25    level: usize,
26    id: CowStr<'a>,
27    /// Its text and markup, less links, which would nest in the entry's
28    /// own, and footnotes, which don't belong in it.
29    content: Vec<Item<'a>>,
30}
31
32impl<'a> Toc<'a> {
33    /// Replaces each list in `items` marked `{:toc}` with a list of links to
34    /// the headings, nested by level, leaving out those with the class
35    /// `no_toc`. The tables hold at most 100,000 items between them, or
36    /// `items.len()` if that's more.
37    pub fn replace_lists(items: Vec<Item<'a>>) -> Vec<Item<'a>> {
38        let budget = items.len().max(100_000);
39        Self::replace_lists_within(items, budget)
40    }
41
42    /// Replaces the lists marked `{:toc}` as long as the tables fit in
43    /// `budget` items, in one pass.
44    fn replace_lists_within(items: Vec<Item<'a>>, mut budget: usize) -> Vec<Item<'a>> {
45        let is_mark = |item: &Item<'a>| {
46            item.attribute_lists.toc && matches!(item.event, Event::Start(Tag::List(_)))
47        };
48        if !items.iter().any(is_mark) {
49            return items;
50        }
51        let toc = Self::new(&items);
52        let size = toc.size();
53        // The lists made into tables, as their first and last items.
54        let mut attribute_lists = Vec::new();
55        let mut i = 0;
56        while i < items.len() {
57            if is_mark(&items[i]) {
58                if size > budget {
59                    // This and every later list stay as written.
60                    break;
61                }
62                budget -= size;
63                let end = Item::end_of(&items, i);
64                attribute_lists.push((i, end));
65                i = end;
66            }
67            i += 1;
68        }
69        let mut out = Vec::with_capacity(items.len() + attribute_lists.len() * size);
70        let mut attribute_lists = attribute_lists.into_iter().peekable();
71        let mut skip_to = None;
72        for (i, item) in items.into_iter().enumerate() {
73            if let Some(end) = skip_to {
74                if i == end {
75                    skip_to = None;
76                }
77                continue;
78            }
79            if let Some(&(start, end)) = attribute_lists.peek()
80                && start == i
81            {
82                attribute_lists.next();
83                toc.write_list(&item, &mut out);
84                if end != i {
85                    skip_to = Some(end);
86                }
87                continue;
88            }
89            out.push(item);
90        }
91        out
92    }
93
94    /// The number of items in each table of contents, none if there are no
95    /// headings to list.
96    fn size(&self) -> usize {
97        if self.entries.is_empty() {
98            return 0;
99        }
100        2 + (0..self.entries.len())
101            .map(|i| {
102                4 + self.entries[i].content.len() + if self.children[i].is_empty() { 0 } else { 2 }
103            })
104            .sum::<usize>()
105    }
106
107    fn new(items: &[Item<'a>]) -> Self {
108        let entries = Entry::all(items);
109        let mut children = vec![Vec::new(); entries.len()];
110        let mut roots = Vec::new();
111        let mut path: Vec<usize> = Vec::new();
112        for (i, entry) in entries.iter().enumerate() {
113            while path
114                .last()
115                .is_some_and(|&p| entries[p].level >= entry.level)
116            {
117                path.pop();
118            }
119            match path.last() {
120                Some(&parent) => children[parent].push(i),
121                None => roots.push(i),
122            }
123            path.push(i);
124        }
125        Self {
126            entries,
127            children,
128            roots,
129        }
130    }
131
132    /// Writes the table of contents standing in for the list `list` starts,
133    /// or nothing, as kramdown does, if there are no headings to list.
134    fn write_list(&self, list: &Item<'a>, out: &mut Vec<Item<'a>>) {
135        let Event::Start(Tag::List(start)) = list.event else {
136            return;
137        };
138        if self.roots.is_empty() {
139            return;
140        }
141        out.push(Item {
142            attribute_lists: list.attribute_lists.clone(),
143            ..Item::new(Event::Start(Tag::List(start)), list.range.clone())
144        });
145        self.write_entries(&self.roots, start, &list.range, out);
146        out.push(Item::new(
147            Event::End(TagEnd::List(start.is_some())),
148            list.range.clone(),
149        ));
150    }
151
152    /// Writes an item linking to each of `entries`, with a list of the
153    /// entries under it, numbered from `start` if the table is numbered.
154    fn write_entries(
155        &self,
156        entries: &[usize],
157        start: Option<u64>,
158        range: &Range<usize>,
159        out: &mut Vec<Item<'a>>,
160    ) {
161        let item = |event| Item::new(event, range.clone());
162        for &i in entries {
163            let entry = &self.entries[i];
164            out.push(item(Event::Start(Tag::Item)));
165            out.push(item(Event::Start(Tag::Link {
166                link_type: LinkType::Inline,
167                dest_url: format!("#{}", entry.id).into(),
168                title: CowStr::Borrowed(""),
169                id: CowStr::Borrowed(""),
170            })));
171            out.extend(entry.content.iter().cloned());
172            out.push(item(Event::End(TagEnd::Link)));
173            if !self.children[i].is_empty() {
174                out.push(item(Event::Start(Tag::List(start))));
175                self.write_entries(&self.children[i], start, range, out);
176                out.push(item(Event::End(TagEnd::List(start.is_some()))));
177            }
178            out.push(item(Event::End(TagEnd::Item)));
179        }
180    }
181}
182
183impl<'a> Entry<'a> {
184    /// The headings in `items` with an id and without the class `no_toc`.
185    fn all(items: &[Item<'a>]) -> Vec<Self> {
186        let mut entries = Vec::new();
187        for (i, item) in items.iter().enumerate() {
188            let Event::Start(Tag::Heading {
189                level,
190                id: Some(id),
191                ..
192            }) = &item.event
193            else {
194                continue;
195            };
196            if item.attribute_lists.attributes.has_class("no_toc") {
197                continue;
198            }
199            let content = items[i + 1..Item::end_of(items, i)]
200                .iter()
201                .filter(|item| {
202                    !matches!(
203                        item.event,
204                        Event::Start(Tag::Link { .. })
205                            | Event::End(TagEnd::Link)
206                            | Event::FootnoteReference(_)
207                    )
208                })
209                .cloned()
210                .collect();
211            entries.push(Self {
212                level: *level as usize,
213                id: id.clone(),
214                content,
215            });
216        }
217        entries
218    }
219}

227    /// The items `text` reads as with `extensions`, tables of contents
228    /// made.
229    fn toc(text: &str, extensions: Extensions) -> Vec<Item<'_>> {
230        Toc::replace_lists(Reader::new(text, Options::from(extensions), &[]).collect())
231    }
232
233    /// The lists and links in `items` before the first heading, as
234    /// HTML-like text.
235    fn outline(items: &[Item<'_>]) -> String {
236        items
237            .iter()
238            .take_while(|item| !matches!(item.event, Event::Start(Tag::Heading { .. })))
239            .map(|item| match &item.event {
240                Event::Start(Tag::List(None)) => "<ul>".into(),
241                Event::End(TagEnd::List(false)) => "</ul>".into(),
242                Event::Start(Tag::List(Some(start))) => format!("<ol start={start}>"),
243                Event::End(TagEnd::List(true)) => "</ol>".into(),
244                Event::Start(Tag::Item) => "<li>".into(),
245                Event::End(TagEnd::Item) => "</li>".into(),
246                Event::Start(Tag::Link { dest_url, .. }) => format!("<a {dest_url}>"),
247                Event::End(TagEnd::Link) => "</a>".into(),
248                Event::Start(Tag::Paragraph) => "<p>".into(),
249                Event::End(TagEnd::Paragraph) => "</p>".into(),
250                Event::Text(text) => text.to_string(),
251                event => panic!("unexpected {event:?}"),
252            })
253            .collect()
254    }

256    #[test]
257    fn toc_replace_lists_puts_entries_under_the_last_heading_of_a_higher_level() {
258        let items = toc("* x\n{:toc}\n\n## A\n# B\n### C\n## D\n", Extensions::all());
259        assert_eq!(
260            outline(&items),
261            "<ul><li><a #a>A</a></li><li><a #b>B</a><ul><li><a #c>C</a></li><li><a #d>D</a></li></ul></li></ul>"
262        );
263    }

265    #[test]
266    fn toc_replace_lists_keeps_the_lists_numbering_and_attributes() {
267        let items = toc("3. x\n{:toc .t}\n\n# A\n", Extensions::all());
268        assert_eq!(outline(&items), "<ol start=3><li><a #a>A</a></li></ol>");
269        assert!(items[0].attribute_lists.attributes.has_class("t"));
270    }

272    #[test]
273    fn toc_replace_lists_leaves_out_headings_without_ids() {
274        let items = toc(
275            "* x\n{:toc}\n\n# A\n# B {#b}\n",
276            Extensions::all() - Extensions::ENABLE_AUTO_IDS,
277        );
278        assert_eq!(outline(&items), "<ul><li><a #b>B</a></li></ul>");
279        // With no headings, there's no table, nor the list.
280        let items = toc("* x\n{:toc}\n\np\n", Extensions::all());
281        assert_eq!(outline(&items), "<p>p</p>");
282    }

290    #[test]
291    fn toc_replace_lists_within_leaves_tables_past_the_budget_as_written() {
292        let text = "* x\n{:toc}\n\np\n\n* x\n{:toc}\n\np\n\n* x\n{:toc}\n\n# A\n";
293        let read = || Reader::new(text, Options::from(Extensions::all()), &[]).collect();
294        let made = "<ul><li><a #a>A</a></li></ul>";
295        let written = "<ul><li>x</li></ul>";
296        let p = "<p>p</p>";
297        // Each table is 7 items: the list, the item, the link and its text.
298        let items = Toc::replace_lists_within(read(), 14);
299        assert_eq!(outline(&items), [made, p, made, p, written].concat());
300        // One short of the second table leaves it, and the third, alone.
301        let items = Toc::replace_lists_within(read(), 13);
302        assert_eq!(outline(&items), [made, p, written, p, written].concat());
303        let items = Toc::replace_lists_within(read(), 6);
304        assert_eq!(outline(&items), [written, p, written, p, written].concat());
305        // The heading after the lists is kept either way.
306        assert!(matches!(
307            &items[items.len() - 2].event,
308            Event::Text(text) if &**text == "A"
309        ));
310    }

312    #[test]
313    fn toc_replace_lists_budget_is_at_least_100_000_items() {
314        // 300 tables of 300 headings would be 450,600 items, far more than
315        // the document's, so only as many tables as fit in 100,000 are made.
316        let text = "* x\n{:toc}\n\n# h\n\n".repeat(300);
317        let items = toc(&text, Extensions::all());
318        let links = items
319            .iter()
320            .filter(|item| matches!(item.event, Event::Start(Tag::Link { .. })))
321            .count();
322        let size = 2 + 300 * 5;
323        assert_eq!(links, 100_000 / size * 300);
324    }
