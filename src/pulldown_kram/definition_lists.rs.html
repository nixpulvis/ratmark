1//! kramdown's definition lists, from pulldown-cmark's: each line before the
2//! definitions is a term of its own, and a definition's first paragraph is
3//! bare unless a blank line comes before the definition.
4
5use pulldown_cmark::{Event, Tag, TagEnd};
6
7use crate::read::Item;
8use crate::text::{self, Lines};
9
10/// The definition lists of a document, read as kramdown reads them.
11pub(crate) struct DefinitionLists<'t> {
12    text: &'t str,
13}
14
15impl<'t> DefinitionLists<'t> {
16    pub fn new(text: &'t str) -> Self {
17        Self { text }
18    }
19
20    /// Rewrites the definition lists in `items`, read from this text.
21    pub fn rewrite(&self, items: &mut Vec<Item<'_>>) {
22        let mut i = 0;
23        while i < items.len() {
24            match items[i].event {
25                Event::Start(Tag::DefinitionListTitle) => Self::split_terms(items, i),
26                Event::Start(Tag::DefinitionListDefinition)
27                    if !self.blank_line_before(items[i].range.start) =>
28                {
29                    Self::unwrap_first_paragraph(items, i);
30                }
31                _ => {}
32            }
33            i += 1;
34        }
35    }
36
37    /// Ends the term starting at `title` at each line break in it, and
38    /// starts another.
39    fn split_terms(items: &mut Vec<Item<'_>>, title: usize) {
40        let mut end = Item::end_of(items, title);
41        let range = items[title].range.clone();
42        let mut depth = 0usize;
43        let mut i = title + 1;
44        while i < end {
45            match items[i].event {
46                Event::Start(_) => depth += 1,
47                Event::End(_) => depth -= 1,
48                Event::SoftBreak if depth == 0 => {
49                    let line = items[i].range.clone();
50                    items[i] = Item::new(
51                        Event::End(TagEnd::DefinitionListTitle),
52                        range.start..line.start,
53                    );
54                    items.insert(
55                        i + 1,
56                        Item::new(Event::Start(Tag::DefinitionListTitle), line.end..range.end),
57                    );
58                    i += 1;
59                    end += 1;
60                }
61                _ => {}
62            }
63            i += 1;
64        }
65    }
66
67    /// Takes the paragraph that starts the definition at `definition` out
68    /// of it, giving the definition its attributes.
69    fn unwrap_first_paragraph(items: &mut Vec<Item<'_>>, definition: usize) {
70        let paragraph = definition + 1;
71        if !matches!(
72            items.get(paragraph).map(|item| &item.event),
73            Some(Event::Start(Tag::Paragraph))
74        ) {
75            return;
76        }
77        let end = Item::end_of(items, paragraph);
78        let attribute_lists = std::mem::take(&mut items[paragraph].attribute_lists);
79        items[definition].attribute_lists.extend(attribute_lists);
80        items.remove(end);
81        items.remove(paragraph);
82    }
83
84    /// Whether the line before the one `offset` is on is blank, or blank
85    /// but for quote markers.
86    fn blank_line_before(&self, offset: usize) -> bool {
87        let lines = Lines(self.text);
88        lines
89            .previous(lines.start(offset))
90            .is_some_and(|before| text::is_blank(lines.rest_of_line(before)))
91    }
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97    use pulldown_cmark::{Options, Parser};
98
99    /// The items `text` reads as, with pulldown-cmark's definition lists.
100    fn items(text: &str) -> Vec<Item<'_>> {
101        Parser::new_ext(text, Options::ENABLE_DEFINITION_LIST)
102            .into_offset_iter()
103            .map(|(event, range)| Item::new(event, range))
104            .collect()
105    }
106
107    /// `items` as HTML-like text.
108    fn outline(items: &[Item<'_>]) -> String {
109        items
110            .iter()
111            .map(|item| match &item.event {
112                Event::Start(Tag::DefinitionList) => "<dl>",
113                Event::End(TagEnd::DefinitionList) => "</dl>",
114                Event::Start(Tag::DefinitionListTitle) => "<dt>",
115                Event::End(TagEnd::DefinitionListTitle) => "</dt>",
116                Event::Start(Tag::DefinitionListDefinition) => "<dd>",
117                Event::End(TagEnd::DefinitionListDefinition) => "</dd>",
118                Event::Start(Tag::Paragraph) => "<p>",
119                Event::End(TagEnd::Paragraph) => "</p>",
120                Event::Start(Tag::Emphasis) => "<em>",
121                Event::End(TagEnd::Emphasis) => "</em>",
122                Event::SoftBreak => "\n",
123                Event::Text(text) => text,
124                event => panic!("unexpected {event:?}"),
125            })
126            .collect()
127    }
128
129    #[test]
130    fn definition_lists_rewrite_splits_terms_only_outside_spans() {
131        let text = "a\n*b\nc*\nd\n: e\n";
132        let mut items = items(text);
133        DefinitionLists::new(text).rewrite(&mut items);
134        assert_eq!(
135            outline(&items),
136            "<dl><dt>a</dt><dt><em>b\nc</em></dt><dt>d</dt><dd>e</dd></dl>"
137        );
138    }
139
140    #[test]
141    fn definition_lists_rewrite_gives_a_bare_definition_its_paragraphs_attributes() {
142        // Loose, as pulldown-cmark only gives a loose list's definitions
143        // paragraphs.
144        for (text, bare) in [("a\n: b\n\n  c\n", true), ("a\n\n: b\n\n  c\n", false)] {
145            let mut items = items(text);
146            let paragraph = items
147                .iter()
148                .position(|item| matches!(item.event, Event::Start(Tag::Paragraph)))
149                .unwrap();
150            items[paragraph]
151                .attribute_lists
152                .attributes
153                .classes
154                .push("x".into());
155            DefinitionLists::new(text).rewrite(&mut items);
156            let definition = items
157                .iter()
158                .find(|item| matches!(item.event, Event::Start(Tag::DefinitionListDefinition)))
159                .unwrap();
160            assert_eq!(
161                definition.attribute_lists.attributes.has_class("x"),
162                bare,
163                "{text:?}"
164            );
165            let paragraphs = items
166                .iter()
167                .filter(|item| matches!(item.event, Event::Start(Tag::Paragraph)))
168                .count();
169            assert_eq!(paragraphs, if bare { 1 } else { 2 }, "{text:?}");
170        }
171    }
172
173    #[test]
174    fn definition_lists_blank_line_before_allows_spaces_and_quote_markers() {
175        let attribute_lists = DefinitionLists::new("a\n>\n  \nb\nc");
176        // The first line has none before it.
177        assert!(!attribute_lists.blank_line_before(0));
178        assert!(!attribute_lists.blank_line_before(2));
179        assert!(attribute_lists.blank_line_before(4));
180        assert!(attribute_lists.blank_line_before(7));
181        // From anywhere on the line.
182        assert!(attribute_lists.blank_line_before(8));
183        assert!(!attribute_lists.blank_line_before(9));
184    }
