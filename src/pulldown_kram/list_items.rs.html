1//! kramdown's list items, from pulldown-cmark's: whether an item's first
2//! paragraph is bare is decided for each item, not for the whole list.
3
4use std::ops::Range;
5
6use pulldown_cmark::{Event, Tag, TagEnd};
7
8use crate::read::Item;
9use crate::text::{self, Lines};
10
11/// The lists of a document, read as kramdown reads them.
12pub(crate) struct ListItems<'t> {
13    text: &'t str,
14}
15
16/// The paragraph an item starts with.
17struct First {
18    /// The index of its first item, and of its last.
19    items: Range<usize>,
20    /// Whether it's wrapped in a paragraph's start and end.
21    wrapped: bool,
22    /// Where it is in the text.
23    range: Range<usize>,
24}
25
26impl<'t> ListItems<'t> {
27    pub fn new(text: &'t str) -> Self {
28        Self { text }
29    }
30
31    /// Wraps the first paragraph of each item in `items` that kramdown
32    /// wraps, and unwraps each it leaves bare, giving the item the
33    /// paragraph's attributes.
34    pub fn rewrite(&self, items: &mut Vec<Item<'_>>) {
35        // Inner lists first, so the indices of those around them hold.
36        for list in (0..items.len()).rev() {
37            if matches!(items[list].event, Event::Start(Tag::List(_))) {
38                self.rewrite_list(items, list);
39            }
40        }
41    }
42
43    fn rewrite_list(&self, items: &mut Vec<Item<'_>>, list: usize) {
44        let end = Item::end_of(items, list);
45        let mut entries = Vec::new();
46        let mut i = list + 1;
47        while i < end {
48            let item_end = Item::end_of(items, i);
49            entries.push((i, item_end));
50            i = item_end + 1;
51        }
52        // For each item, its first paragraph, and whether kramdown leaves it
53        // bare.
54        let mut firsts: Vec<Option<(First, bool)>> = Vec::with_capacity(entries.len());
55        for (n, &(item, item_end)) in entries.iter().enumerate() {
56            let Some(first) = Self::first(items, item, item_end) else {
57                firsts.push(None);
58                continue;
59            };
60            let last = n + 1 == entries.len();
61            let only = first.items.end + 1 + usize::from(first.wrapped) == item_end;
62            let blank = self.blank_line_after(first.range.end);
63            // kramdown's rule: bare unless a blank line follows, but for the
64            // last item's only paragraph, and then only if an item before
65            // it is bare too, or doesn't start with a paragraph.
66            let first_bare = !blank || last && only;
67            let after_others = !last
68                || entries.len() == 1
69                || firsts
70                    .iter()
71                    .any(|other| !matches!(other, Some((_, false))));
72            firsts.push(Some((first, first_bare && after_others)));
73        }
74        for (first, bare) in firsts.into_iter().rev().flatten() {
75            match (first.wrapped, bare) {
76                (true, true) => {
77                    let paragraph = first.items.start - 1;
78                    let attribute_lists = std::mem::take(&mut items[paragraph].attribute_lists);
79                    items[paragraph - 1].attribute_lists.extend(attribute_lists);
80                    items.remove(first.items.end + 1);
81                    items.remove(paragraph);
82                }
83                (false, false) => {
84                    let range = first.range;
85                    items.insert(
86                        first.items.end + 1,
87                        Item::new(Event::End(TagEnd::Paragraph), range.clone()),
88                    );
89                    items.insert(
90                        first.items.start,
91                        Item::new(Event::Start(Tag::Paragraph), range),
92                    );
93                }
94                _ => {}
95            }
96        }
97    }
98
99    /// The paragraph the item from `item` to `item_end` starts with, if it
100    /// starts with one: wrapped, as in a loose list, or bare, as in a tight
101    /// one.
102    fn first(items: &[Item<'_>], item: usize, item_end: usize) -> Option<First> {
103        let start = item + 1;
104        if matches!(items[start].event, Event::Start(Tag::Paragraph)) {
105            let end = Item::end_of(items, start);
106            return Some(First {
107                items: start + 1..end - 1,
108                wrapped: true,
109                range: items[start].range.clone(),
110            });
111        }
112        let len = items[start..item_end]
113            .iter()
114            .position(|item| !Item::inline(&item.event))
115            .unwrap_or(item_end - start);
116        (len > 0).then(|| First {
117            items: start..start + len - 1,
118            wrapped: false,
119            range: items[start].range.start..items[start + len - 1].range.end,
120        })
121    }
122
123    /// Whether the line after the one `end` ends is blank, or blank but for
124    /// quote markers. The text's end isn't a blank line.
125    fn blank_line_after(&self, end: usize) -> bool {
126        let lines = Lines(self.text);
127        let next = if self.text[..end].ends_with('\n') {
128            end
129        } else {
130            lines.next(end)
131        };
132        next < self.text.len() && text::is_blank(lines.rest_of_line(next))
133    }
134}
