6/// Parsing options for a [`Document`](crate::Document): which kramdown
7/// extensions to enable, plus any extra pulldown-cmark options.
8///
9/// Each part of kramdown's syntax has an [`Extensions`] flag, so
10/// [`Extensions::all()`] matches kramdown, and converts into [`Options`]
11/// with no extra pulldown-cmark options. Some extensions, like tables, are
12/// implemented by pulldown-cmark and turn on its matching option. Those
13/// options are controlled by their extension alone, and are on exactly when
14/// it is, regardless of [`cmark`]. Use [`cmark`] for syntax kramdown doesn't
15/// have, like GitHub's `> [!NOTE]` blockquotes from [`ENABLE_GFM`].
16///
17/// [`cmark`] can't turn on [`ENABLE_TABLES`], [`ENABLE_FOOTNOTES`],
18/// [`ENABLE_STRIKETHROUGH`], [`ENABLE_TASKLISTS`],
19/// [`ENABLE_SMART_PUNCTUATION`], [`ENABLE_DEFINITION_LIST`] or
20/// [`ENABLE_MATH`], which are controlled by the extensions, or
21/// [`ENABLE_HEADING_ATTRIBUTES`], which parses `{…}` after a heading
22/// differently from kramdown. [`ENABLE_OLD_FOOTNOTES`] only has an effect
23/// with [`Extensions::ENABLE_FOOTNOTES`].
24///
25/// [`cmark`]: Self::cmark
26/// [`ENABLE_GFM`]: pulldown_cmark::Options::ENABLE_GFM
27/// [`ENABLE_TABLES`]: pulldown_cmark::Options::ENABLE_TABLES
28/// [`ENABLE_FOOTNOTES`]: pulldown_cmark::Options::ENABLE_FOOTNOTES
29/// [`ENABLE_STRIKETHROUGH`]: pulldown_cmark::Options::ENABLE_STRIKETHROUGH
30/// [`ENABLE_TASKLISTS`]: pulldown_cmark::Options::ENABLE_TASKLISTS
31/// [`ENABLE_SMART_PUNCTUATION`]: pulldown_cmark::Options::ENABLE_SMART_PUNCTUATION
32/// [`ENABLE_DEFINITION_LIST`]: pulldown_cmark::Options::ENABLE_DEFINITION_LIST
33/// [`ENABLE_MATH`]: pulldown_cmark::Options::ENABLE_MATH
34/// [`ENABLE_HEADING_ATTRIBUTES`]: pulldown_cmark::Options::ENABLE_HEADING_ATTRIBUTES
35/// [`ENABLE_OLD_FOOTNOTES`]: pulldown_cmark::Options::ENABLE_OLD_FOOTNOTES
36///
37/// ```
38/// use pulldown_kram::{Document, Extensions, Options};
39/// use pulldown_kram::pulldown_cmark::Options as Cmark;
40///
41/// // Plain kramdown.
42/// let kramdown = Document::new("a | b\n-|-\n", Extensions::all());
43///
44/// // Plus GitHub's `> [!NOTE]` blockquotes.
45/// let more = Options::new(Extensions::all(), Cmark::ENABLE_GFM);
46/// let document = Document::new("> [!NOTE]\n> a\n", more);
47/// ```
48#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
49pub struct Options {
50    /// The kramdown extensions to enable.
51    pub kramdown: Extensions,
52    /// Extra pulldown-cmark options, for syntax kramdown doesn't have.
53    /// Options that belong to an extension are ignored.
54    pub cmark: pulldown_cmark::Options,
55}
56
57impl Options {
58    /// Creates options from kramdown extensions and extra pulldown-cmark
59    /// options.
60    pub fn new(kramdown: Extensions, cmark: pulldown_cmark::Options) -> Self {
61        Self { kramdown, cmark }
62    }
63
64    /// The options pulldown-cmark reads with: those the extensions need,
65    /// and the rest of `cmark`.
66    pub(crate) fn parser_options(&self) -> pulldown_cmark::Options {
67        let read =
68            Extensions::all().cmark_options() | pulldown_cmark::Options::ENABLE_HEADING_ATTRIBUTES;
69        (self.cmark - read) | self.kramdown.cmark_options()
70    }
71}
72
73impl From<Extensions> for Options {
74    /// Enables the `kramdown` extensions, with no extra pulldown-cmark
75    /// options.
76    fn from(kramdown: Extensions) -> Self {
77        Self {
78            kramdown,
79            cmark: pulldown_cmark::Options::empty(),
80        }
81    }
82}
83
84impl Default for Options {
85    /// Every extension, matching kramdown, with no extra pulldown-cmark
86    /// options.
87    fn default() -> Self {
88        Self::from(Extensions::all())
89    }
90}

92bitflags! {
93    /// The parts of kramdown's syntax that go beyond CommonMark, each of
94    /// which can be turned off.
95    ///
96    /// With every extension on, as in [`Extensions::all()`], documents are
97    /// parsed the way kramdown parses them. Some extensions are parsed by
98    /// pulldown-cmark itself, and say which of its options they turn on. See
99    /// [`Options`] for how those interact with [`Options::cmark`].
100    ///
101    /// The flags come in groups:
102    ///
103    /// - Syntax pulldown-cmark also reads, from tables to math, which turn
104    ///   on its option for it.
105    /// - Attribute lists, and the ids and classes that come from them.
106    /// - kramdown's own block and span syntax.
107    /// - Where kramdown reads CommonMark's syntax differently.
108    ///
109    /// [`ENABLE_TABLES_OF_CONTENTS`](Self::ENABLE_TABLES_OF_CONTENTS) makes
110    /// [`Document::events`](crate::Document::events) collect every event
111    /// before returning the first, since a table can link to headings after
112    /// it. [`Extensions::streamable()`] leaves it out, so events are returned
113    /// a block at a time. Any other extension can be turned off by removing
114    /// it from [`Extensions::all()`]:
115    ///
116    /// ```
117    /// use pulldown_kram::{Document, Extensions, pulldown_cmark::Event};
118    ///
119    /// // Everything except typography, so quotes stay straight.
120    /// let extensions = Extensions::all() - Extensions::ENABLE_TYPOGRAPHY;
121    /// let document = Document::new("\"a\"\n", extensions);
122    /// let text: Vec<_> = document
123    ///     .events()
124    ///     .filter_map(|(event, _)| match event {
125    ///         Event::Text(text) => Some(text),
126    ///         _ => None,
127    ///     })
128    ///     .collect();
129    /// assert_eq!(text, ["\"a\"".into()]);
130    /// ```
131    #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
132    pub struct Extensions: u32 {
133        // Syntax pulldown-cmark also reads.
134
135        /// Tables with a header row. Turns on pulldown-cmark's
136        /// `ENABLE_TABLES`.
137        const ENABLE_TABLES = 1 << 0;
138        /// Footnotes, `[^1]`, with block content in their definitions. Turns
139        /// on pulldown-cmark's `ENABLE_FOOTNOTES`.
140        const ENABLE_FOOTNOTES = 1 << 1;
141        /// `~~Strikethrough~~`, from kramdown's GFM parser. Turns on
142        /// pulldown-cmark's `ENABLE_STRIKETHROUGH`.
143        const ENABLE_STRIKETHROUGH = 1 << 2;
144        /// Task list items, `* [x] done`, from kramdown's GFM parser, with
145        /// its classes: `task-list` on the list, and `task-list-item` on
146        /// each item from the first task onward. Turns on pulldown-cmark's
147        /// `ENABLE_TASKLISTS`.
148        const ENABLE_TASK_LISTS = 1 << 3;
149        /// Dashes, ellipses and smart quotes. As in kramdown, header ids
150        /// leave these characters out, so ids only match kramdown's with this
151        /// on. Turns on pulldown-cmark's `ENABLE_SMART_PUNCTUATION`.
152        const ENABLE_TYPOGRAPHY = 1 << 4;
153        /// Definition lists: a term followed by `: definition` lines. Turns
154        /// on pulldown-cmark's `ENABLE_DEFINITION_LIST`.
155        const ENABLE_DEFINITION_LISTS = 1 << 5;
156        /// Math between `$$`: a block when it's alone in its paragraph,
157        /// inline otherwise. As in kramdown, a single `$` is plain text.
158        /// Turns on pulldown-cmark's `ENABLE_MATH`.
159        const ENABLE_MATH = 1 << 6;
160
161        // Attribute lists, and what comes from them.
162
163        /// Block attribute lists, like `{:.note}` on the line after a
164        /// paragraph, or alone between blocks to apply to the next one.
165        const ENABLE_BLOCK_ATTRIBUTES = 1 << 7;
166        /// Span attribute lists directly after a span, like `*a*{:.x}`, and
167        /// the empty list, `{::}`, as a separator between spans.
168        const ENABLE_SPAN_ATTRIBUTES = 1 << 8;
169        /// Attribute lists at the start of a list item, like `* {:.x} a`.
170        const ENABLE_ITEM_ATTRIBUTES = 1 << 9;
171        /// Header ids after a heading's text, `# Hello {#hi}`.
172        const ENABLE_HEADER_IDS = 1 << 10;
173        /// Automatic ids for headings without one, generated the same way as
174        /// kramdown's GFM parser. kramdown's `auto_ids` option.
175        const ENABLE_AUTO_IDS = 1 << 11;
176        /// A code block's language from a `language-` class, like
177        /// `{: .language-ruby}`, when its fence doesn't give one. The class
178        /// comes from a block attribute list, so this needs
179        /// `ENABLE_BLOCK_ATTRIBUTES`.
180        const ENABLE_LANGUAGE_CLASSES = 1 << 12;
181        /// Lists marked `{:toc}` turned into tables of contents, linking to
182        /// every heading that has an id and doesn't have the class `no_toc`.
183        /// The marker is a block attribute list, so this needs
184        /// `ENABLE_BLOCK_ATTRIBUTES`.
185        ///
186        /// Every event is collected before the first is returned,
187        /// so tables can link to headings after them. Each table lists every
188        /// heading, so output could grow with the square of the document's
189        /// size. To bound it, tables of contents produce at most 100,000
190        /// events or as many as the document has, whichever is larger; marked
191        /// lists past that limit are left as written.
192        const ENABLE_TABLES_OF_CONTENTS = 1 << 13;
193
194        // kramdown's own syntax.
195
196        /// The end-of-block marker, a line with only `^`, closing the blocks
197        /// open at that point, like a list or a code block. It produces no
198        /// events.
199        const ENABLE_END_OF_BLOCK_MARKERS = 1 << 14;
200        /// Hard line breaks with two backslashes at the end of a line, as
201        /// well as with two spaces.
202        const ENABLE_BACKSLASH_LINE_BREAKS = 1 << 15;
203        /// Tables without a header row, like a lone `| a | b |` line, and a
204        /// `|` in code in a table cell kept in that cell instead of starting
205        /// a new one. Also accepts a separator line with fewer cells than
206        /// the header, or `+` between its cells. pulldown-cmark parses the
207        /// tables, so this needs `ENABLE_TABLES`.
208        const ENABLE_TABLES_WITHOUT_HEADERS = 1 << 16;
209        /// Abbreviations, `*[HTML]: HyperText Markup Language`. Each use in
210        /// the text becomes an `<abbr>` with its title.
211        const ENABLE_ABBREVIATIONS = 1 << 17;
212        /// Markdown inside HTML elements marked `markdown="1"`. The
213        /// attribute is removed from the HTML.
214        const ENABLE_MARKDOWN_ATTRIBUTE = 1 << 18;
215        /// kramdown's `comment` extension, `{::comment}…{:/}`, as a block or
216        /// span, emitted as an HTML comment.
217        const ENABLE_COMMENTS = 1 << 19;
218        /// kramdown's `nomarkdown` extension, `{::nomarkdown}…{:/}`, as a
219        /// block or span, emitted as raw HTML unless its `type` names another
220        /// of kramdown's converters, like `latex`.
221        const ENABLE_NOMARKDOWN = 1 << 20;
222
223        // Where kramdown reads CommonMark's syntax differently.
224
225        /// HTML blocks that end at the end tag of the element that opened
226        /// them, as in kramdown, with any text after it parsed as Markdown.
227        /// CommonMark's HTML blocks run to the next blank line. A blank line
228        /// before the end tag still ends the block, as in CommonMark.
229        const ENABLE_HTML_BLOCK_ENDS = 1 << 21;
230        /// A line with only a span-level tag, like `<img>`, starts a
231        /// paragraph together with the text on the next line, instead of an
232        /// HTML block. Followed by a blank line, it's still an HTML block,
233        /// where kramdown makes it a paragraph.
234        const ENABLE_SPAN_TAG_PARAGRAPHS = 1 << 22;
235        /// The text of the HTML spans kramdown leaves raw, like `<kbd>`,
236        /// `<code>` and `<var>`, passed through unchanged, without Markdown
237        /// or typography applied inside them.
238        const ENABLE_RAW_HTML_SPANS = 1 << 23;
239        /// Tight and loose list items decided per item, as kramdown does,
240        /// rather than per list: an item's first paragraph is only wrapped
241        /// in a paragraph if a blank line follows it in the item.
242        const ENABLE_LIST_ITEM_PARAGRAPHS = 1 << 24;
243        /// One list for items marked with any mix of `*`, `+` and `-`.
244        /// CommonMark starts a new list whenever the marker changes.
245        const ENABLE_MIXED_LIST_MARKERS = 1 << 25;
246        /// Backslash escapes for only the characters kramdown escapes. A
247        /// backslash before any other character, like `\&`, is kept, where
248        /// CommonMark escapes all ASCII punctuation.
249        const ENABLE_KRAMDOWN_ESCAPES = 1 << 26;
250    }
251}
252
253impl Extensions {
254    /// Every extension except
255    /// [`ENABLE_TABLES_OF_CONTENTS`](Self::ENABLE_TABLES_OF_CONTENTS), so
256    /// [`Document::events`](crate::Document::events) returns events a block at
257    /// a time instead of collecting them all first. Lists marked `{:toc}`
258    /// stay as written.
259    ///
260    /// ```
261    /// use pulldown_kram::{Document, Extensions};
262    ///
263    /// let document = Document::new("* x\n{:toc}\n", Extensions::streamable());
264    /// assert_eq!(document.events().count(), 5);
265    /// ```
266    pub const fn streamable() -> Self {
267        Self::all().difference(Self::ENABLE_TABLES_OF_CONTENTS)
268    }
269
270    /// The extensions pulldown-cmark reads, and its option for each.
271    const CMARK: [(Self, pulldown_cmark::Options); 7] = [
272        (Self::ENABLE_TABLES, pulldown_cmark::Options::ENABLE_TABLES),
273        (
274            Self::ENABLE_FOOTNOTES,
275            pulldown_cmark::Options::ENABLE_FOOTNOTES,
276        ),
277        (
278            Self::ENABLE_STRIKETHROUGH,
279            pulldown_cmark::Options::ENABLE_STRIKETHROUGH,
280        ),
281        (
282            Self::ENABLE_TASK_LISTS,
283            pulldown_cmark::Options::ENABLE_TASKLISTS,
284        ),
285        (
286            Self::ENABLE_TYPOGRAPHY,
287            pulldown_cmark::Options::ENABLE_SMART_PUNCTUATION,
288        ),
289        (
290            Self::ENABLE_DEFINITION_LISTS,
291            pulldown_cmark::Options::ENABLE_DEFINITION_LIST,
292        ),
293        (Self::ENABLE_MATH, pulldown_cmark::Options::ENABLE_MATH),
294    ];
295
296    /// The pulldown-cmark options these extensions are read with.
297    pub(crate) fn cmark_options(self) -> pulldown_cmark::Options {
298        Self::CMARK
299            .into_iter()
300            .filter(|(extension, _)| self.contains(*extension))
301            .fold(pulldown_cmark::Options::empty(), |options, (_, option)| {
302                options | option
303            })
304    }
305
306    /// Whether any kind of attribute list is read.
307    pub(crate) fn attribute_lists(self) -> bool {
308        self.intersects(
309            Self::ENABLE_BLOCK_ATTRIBUTES
310                | Self::ENABLE_SPAN_ATTRIBUTES
311                | Self::ENABLE_ITEM_ATTRIBUTES,
312        )
313    }
314}

316#[cfg(test)]
317mod tests {
318    use super::*;
319    use pulldown_cmark::Options as Cmark;
320
321    #[test]
322    fn options_parser_options_kramdown_needs_nothing_more() {
323        assert_eq!(
324            Options::from(Extensions::all()).parser_options(),
325            Cmark::ENABLE_TABLES
326                | Cmark::ENABLE_FOOTNOTES
327                | Cmark::ENABLE_STRIKETHROUGH
328                | Cmark::ENABLE_TASKLISTS
329                | Cmark::ENABLE_SMART_PUNCTUATION
330                | Cmark::ENABLE_DEFINITION_LIST
331                | Cmark::ENABLE_MATH
332        );
333    }
334
335    #[test]
336    fn options_parser_options_cmark_adds_only_what_kramdown_lacks() {
337        let options = Options::new(Extensions::empty(), Cmark::all()).parser_options();
338        assert!(!options.intersects(Extensions::all().cmark_options()));
339        assert!(!options.contains(Cmark::ENABLE_HEADING_ATTRIBUTES));
340        assert!(options.contains(Cmark::ENABLE_GFM));
341    }
