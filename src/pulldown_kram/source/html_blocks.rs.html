5/// kramdown ends an HTML block at the end tag of the element that starts
6/// it, or at the end of a comment, and reads any text after that as
7/// Markdown. CommonMark runs most HTML blocks to the next blank line. So
8/// this puts a blank line after the end, with a line break first if more
9/// text follows on the same line.
10pub(crate) struct HtmlBlocks;
11
12impl HtmlBlocks {

67    /// `markdown`, with each HTML block ended where kramdown ends it, and
68    /// the offsets of the lines added.
69    pub fn rewrite(markdown: &str) -> (Cow<'_, str>, Offsets) {
70        if !markdown.contains('<') {
71            return (Cow::Borrowed(markdown), Offsets::default());
72        }
73        let mut out = Rewrite::of(markdown);
74        let lines: Vec<&str> = markdown.split_inclusive('\n').collect();
75        let mut fence = Fence::default();
76        // Whether the lines are in an HTML block this leaves as it is.
77        let mut html = false;
78        // What's left of a line after an HTML block ending on it.
79        let mut rest: Option<&str> = None;
80        let mut i = 0;
81        loop {
82            let line = match rest.take() {
83                Some(line) => line,
84                None if i < lines.len() => {
85                    i += 1;
86                    lines[i - 1]
87                }
88                None => break,
89            };
90            if fence.read(line) {
91                out.keep(line);
92                continue;
93            }
94            if line.trim().is_empty() {
95                html = false;
96                out.keep(line);
97                continue;
98            }
99            let content = line.trim_start_matches(' ');
100            if html || line.len() - content.len() > 3 || !content.starts_with('<') {
101                out.keep(line);
102                continue;
103            }
104            let Some((last, end)) = Self::end(content, &lines[i..]) else {
105                // Not a block kramdown ends early, or one that never ends:
106                // left for CommonMark.
107                html = true;
108                out.keep(line);
109                continue;
110            };
111            let end_line = if last == 0 { line } else { lines[i + last - 1] };
112            let end = if last == 0 {
113                line.len() - content.len() + end
114            } else {
115                end
116            };
117            if last > 0 {
118                out.keep(line);
119                for line in &lines[i..i + last - 1] {
120                    out.keep(line);
121                }
122                i += last;
123            }
124            out.keep(&end_line[..end]);
125            let after = &end_line[end..];
126            if after.trim().is_empty() {
127                out.keep(after);
128                if !after.ends_with('\n') {
129                    continue;
130                }
131                if lines.get(i).is_some_and(|next| !next.trim().is_empty()) {
132                    out.insert("\n");
133                }
134            } else {
135                out.insert("\n\n");
136                rest = Some(after);
137            }
138        }
139        out.finish(markdown)
140    }
141
142    /// Where the HTML block that starts with `content` ends, if kramdown
143    /// ends it early. That's the index of the line it ends on, where `0` is
144    /// `content`'s line and `1` the first of `after`, and the offset of the
145    /// end in that line, or in `content` for its own line.
146    fn end(content: &str, after: &[&str]) -> Option<(usize, usize)> {
147        if content.starts_with("<!--") {
148            return std::iter::once(content)
149                .chain(after.iter().copied())
150                .enumerate()
151                .find_map(|(n, line)| {
152                    let from = if n == 0 { "<!--".len() } else { 0 };
153                    let at = line[from..].find("-->")?;
154                    Some((n, from + at + "-->".len()))
155                });
156        }
157        let tag = HtmlTag::starting(content).filter(|tag| !tag.closing)?;
158        let element = Self::ELEMENTS
159            .into_iter()
160            .find(|element| element.eq_ignore_ascii_case(tag.name))?;
161        let opening = &content[..content.find('>')? + 1];
162        if opening.contains("markdown=") {
163            return None;
164        }
165        if opening.ends_with("/>") || matches!(element, "hr" | "col") {
166            return Some((0, opening.len()));
167        }
168        let mut depth = 0usize;
169        for (n, line) in std::iter::once(content)
170            .chain(after.iter().copied())
171            .enumerate()
172        {
173            for (at, _) in line.match_indices('<') {
174                let tag = &line[at + 1..];
175                let (closing, tag) = match tag.strip_prefix('/') {
176                    Some(tag) => (true, tag),
177                    None => (false, tag),
178                };
179                let named = tag
180                    .get(..element.len())
181                    .is_some_and(|name| name.eq_ignore_ascii_case(element))
182                    && tag[element.len()..].starts_with([' ', '\t', '\n', '\r', '>', '/']);
183                if !named {
184                    continue;
185                }
186                let Some(close) = line[at..].find('>').map(|close| at + close + 1) else {
187                    continue;
188                };
189                if closing {
190                    depth = depth.saturating_sub(1);
191                    if depth == 0 {
192                        return Some((n, close));
193                    }
194                } else if !line[..close].ends_with("/>") {
195                    depth += 1;
196                }
197            }
198        }
199        None
200    }
201}

203#[cfg(test)]
204mod tests {
205    use super::*;
206
207    #[test]
208    fn html_blocks_rewrite_ends_a_block_at_its_end_tag() {
209        let src = "<DIV>\n<div>a</div>\n</div>\nb\n<p>c</p> <hr> d\n<!-- e --> f\n";
210        let (text, offsets) = HtmlBlocks::rewrite(src);
211        assert_eq!(
212            text,
213            "<DIV>\n<div>a</div>\n</div>\n\nb\n<p>c</p>\n\n <hr>\n\n d\n<!-- e -->\n\n f\n"
214        );
215        offsets.assert_fit(src, &text);
216        // Not span elements, ones never ended, or what a blank line ends.
217        let src = "<span>a</span> b\n<div>\nc\n\n<p>d</p>\n\ne\n";
218        let (text, offsets) = HtmlBlocks::rewrite(src);
219        assert!(matches!(text, Cow::Borrowed(_)));
220        assert!(offsets.is_empty());
221    }
