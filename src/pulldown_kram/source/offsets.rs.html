4/// Maps offsets in rewritten text back to the text it was rewritten from.
5///
6/// Text outside the edits is the same in both, so its offsets only move by
7/// what the edits before it put in or took out. A range starting inside an
8/// edit's text starts where the text it replaced starts, and one ending
9/// inside it ends where that text ends, so a range takes in all of what it
10/// covers part of. Text put in where there was none maps to the point it
11/// went in at. So an ordered range on char boundaries in the rewritten text
12/// maps to an ordered range on char boundaries in the original.
13///
14/// Text left as it was has no edits, and costs nothing.
15#[derive(Debug, Clone, Default, PartialEq, Eq)]
16pub(crate) struct Offsets {
17    /// The edits, in order, none touching another.
18    edits: Vec<Edit>,
19}
20
21/// Text put in place of some of the original, or of none of it.
22#[derive(Debug, Clone, PartialEq, Eq)]
23struct Edit {
24    /// Where the new text is in the rewritten text. Never empty.
25    rewritten: Range<usize>,
26    /// Where the text it replaced was in the original.
27    original: Range<usize>,
28}
29
30impl Offsets {
31    /// Whether nothing was rewritten.
32    pub fn is_empty(&self) -> bool {
33        self.edits.is_empty()
34    }
35
36    /// Where `range`, in the rewritten text, was in the original.
37    pub fn original(&self, range: Range<usize>) -> Range<usize> {
38        self.to_original(range.start, false)..self.to_original(range.end, true)
39    }
40
41    /// These offsets followed by `later`'s, for when the rewritten text is
42    /// rewritten again.
43    pub fn then(self, later: Offsets) -> Offsets {
44        if self.edits.is_empty() {
45            return later;
46        }
47        if later.edits.is_empty() {
48            return self;
49        }
50        // What either rewrite changed of the text between them, joined
51        // where it overlaps or touches.
52        let mut changed: Vec<Range<usize>> = self
53            .edits
54            .iter()
55            .map(|edit| edit.rewritten.clone())
56            .chain(later.edits.iter().map(|edit| edit.original.clone()))
57            .collect();
58        changed.sort_by_key(|span| span.start);
59        changed.dedup_by(|span, last| {
60            let touches = span.start <= last.end;
61            if touches {
62                last.end = last.end.max(span.end);
63            }
64            touches
65        });
66        // Neither rewrite's edits are partly in a span, so its ends map
67        // exactly both ways.
68        let edits = changed
69            .into_iter()
70            .map(|span| Edit {
71                rewritten: later.to_rewritten(span.start, false)
72                    ..later.to_rewritten(span.end, true),
73                original: self.to_original(span.start, false)..self.to_original(span.end, true),
74            })
75            .collect();
76        Offsets { edits }
77    }
78
79    /// Where `at`, in the rewritten text, was in the original. Inside an
80    /// edit's text, that's where the text it replaced ends, if `at` ends a
81    /// range, or else where it starts.
82    fn to_original(&self, at: usize, end: bool) -> usize {
83        let before = self.edits.partition_point(|edit| edit.rewritten.start < at);
84        let Some(edit) = before.checked_sub(1).map(|i| &self.edits[i]) else {
85            return at;
86        };
87        if at < edit.rewritten.end {
88            return if end {
89                edit.original.end
90            } else {
91                edit.original.start
92            };
93        }
94        edit.original.end + (at - edit.rewritten.end)
95    }
96
97    /// Where `at`, in the original and not inside a replaced text, is in
98    /// the rewritten text. Text put in at `at` comes before it if `at` ends
99    /// a range, and after it otherwise.
100    fn to_rewritten(&self, at: usize, end: bool) -> usize {
101        let before = self
102            .edits
103            .partition_point(|edit| edit.original.start < at || end && edit.original.start == at);
104        match before.checked_sub(1).map(|i| &self.edits[i]) {
105            Some(edit) => edit.rewritten.end + (at - edit.original.end),
106            None => at,
107        }
108    }
109
110    /// Where `at`, in the original and not inside a replaced text, is in
111    /// the rewritten text, after any text put in at it.
112    pub fn moved(&self, at: usize) -> usize {
113        self.to_rewritten(at, true)
114    }
115
116    /// Records `edit`, made after the others, joined onto the last if they
117    /// touch.
118    fn push(&mut self, edit: Edit) {
119        debug_assert!(!edit.rewritten.is_empty());
120        match self.edits.last_mut() {
121            // Text between edits is the same length in both, so they touch
122            // in the original exactly when they do in the rewritten text.
123            Some(last) if last.rewritten.end == edit.rewritten.start => {
124                last.rewritten.end = edit.rewritten.end;
125                last.original.end = edit.original.end;
126            }
127            _ => self.edits.push(edit),
128        }
129    }
130
131    /// Whether `range`, in the original, overlaps or touches text that was
132    /// replaced, or a point text was put in at.
133    #[cfg(test)]
134    pub fn touches(&self, range: &Range<usize>) -> bool {
135        self.edits
136            .iter()
137            .any(|edit| edit.original.start <= range.end && range.start <= edit.original.end)
138    }
139
140    /// What the first `needle` in `rewritten` was in `src`.
141    #[cfg(test)]
142    pub fn was<'s>(&self, src: &'s str, rewritten: &str, needle: &str) -> &'s str {
143        let start = rewritten
144            .find(needle)
145            .expect("needle in the rewritten text");
146        &src[self.original(start..start + needle.len())]
147    }
148
149    /// Checks every range on char boundaries in `rewritten` maps to an
150    /// ordered range on char boundaries in `src`.
151    #[cfg(test)]
152    pub fn assert_fit(&self, src: &str, rewritten: &str) {
153        let boundaries: Vec<usize> = (0..=rewritten.len())
154            .filter(|&i| rewritten.is_char_boundary(i))
155            .collect();
156        for (i, &start) in boundaries.iter().enumerate() {
157            for &end in &boundaries[i..] {
158                let range = self.original(start..end);
159                assert!(
160                    src.get(range.clone()).is_some(),
161                    "{start}..{end} maps to {range:?} in {src:?}"
162                );
163            }
164        }
165        assert_eq!(self.original(0..rewritten.len()), 0..src.len());
166    }
167}

169/// A rewrite in progress: the text written so far, and its edits.
170#[derive(Default)]
171pub(super) struct Rewrite {
172    pub(super) out: String,
173    offsets: Offsets,
174    /// How much of the original has been read.
175    pub(super) read: usize,
176}
177
178impl Rewrite {
179    /// A rewrite of `original`, with nothing read yet.
180    pub(super) fn of(original: &str) -> Self {
181        Self {
182            // Room for the few lines the rewrites put in.
183            out: String::with_capacity(original.len() + 64),
184            ..Self::default()
185        }
186    }
187
188    /// Copies `text`, the next part of the original, unchanged.
189    pub(super) fn keep(&mut self, text: &str) {
190        self.out.push_str(text);
191        self.read += text.len();
192    }
193
194    /// Inserts `text` without replacing any of the original.
195    pub(super) fn insert(&mut self, text: &str) {
196        self.replace(0, text);
197    }
198
199    /// Writes `text` in place of the next `len` bytes of the original.
200    pub(super) fn replace(&mut self, len: usize, text: &str) {
201        let start = self.out.len();
202        self.out.push_str(text);
203        self.offsets.push(Edit {
204            rewritten: start..self.out.len(),
205            original: self.read..self.read + len,
206        });
207        self.read += len;
208    }
209
210    /// The rewritten text and its offsets, or `original` itself, borrowed,
211    /// if nothing changed.
212    pub(super) fn finish(self, original: &str) -> (Cow<'_, str>, Offsets) {
213        debug_assert_eq!(self.read, original.len());
214        if self.offsets.is_empty() {
215            (Cow::Borrowed(original), Offsets::default())
216        } else {
217            (Cow::Owned(self.out), self.offsets)
218        }
219    }
220}

227    /// `src` rewritten by two rewrites, and its offsets.
228    fn rewrite(src: &str) -> (String, Offsets) {
229        let (separated, first) = MarkdownElements::rewrite(src);
230        let (joined, second) = SpanTags::rewrite(&separated);
231        (joined.into_owned(), first.then(second))
232    }
233
234    #[test]
235    fn offsets_original_skips_added_blank_lines() {
236        let src = "<div markdown=\"1\">\nä *ö*\n</div>\nü\n";
237        let (text, offsets) = MarkdownElements::rewrite(src);
238        assert_eq!(text, "<div markdown=\"1\">\n\nä *ö*\n\n</div>\nü\n");
239        for needle in ["<div markdown=\"1\">\n", "ä *ö*\n", "</div>\n", "ü\n"] {
240            assert_eq!(offsets.was(src, &text, needle), needle);
241        }
242        // An added line maps to the point it went in at, starting or ending
243        // a range.
244        let open = "<div markdown=\"1\">\n".len();
245        assert_eq!(offsets.original(open..open + 1), open..open);
246        assert_eq!(offsets.was(src, &text, "\n\nä"), "\nä");
247        assert_eq!(offsets.was(src, &text, "*\n\n"), "*\n");
248        offsets.assert_fit(src, &text);
249    }
250
251    #[test]
252    fn offsets_original_takes_in_joined_line_ends() {
253        let src = "<img alt=\"é\" />  \r\nä\n<b>\nö\n";
254        let (text, offsets) = SpanTags::rewrite(src);
255        assert_eq!(text, "<img alt=\"é\" /> ä\n<b> ö\n");
256        for needle in ["<img alt=\"é\" />", "ä\n", "<b>", "ö\n"] {
257            assert_eq!(offsets.was(src, &text, needle), needle);
258        }
259        // The space in place of a line's end maps to all of it.
260        assert_eq!(offsets.was(src, &text, "/> ä"), "/>  \r\nä");
261        assert_eq!(offsets.was(src, &text, " ö"), "\nö");
262        offsets.assert_fit(src, &text);
263    }

265    #[test]
266    fn offsets_then_follows_both_rewrites() {
267        let src = "<div markdown=\"1\">\n<b>\nü *x*\n</div>\n<i>  \ny\n";
268        let (text, offsets) = rewrite(src);
269        assert_eq!(text, "<div markdown=\"1\">\n\n<b> ü *x*\n\n</div>\n<i> y\n");

271            "<div markdown=\"1\">\n",

273            "ü *x*\n",
274            "</div>\n",

278            assert_eq!(offsets.was(src, &text, needle), needle);
279        }
280        assert_eq!(offsets.was(src, &text, "\n\n<b> ü"), "\n<b>\nü");
281        assert_eq!(offsets.was(src, &text, "<i> y"), "<i>  \ny");
282        offsets.assert_fit(src, &text);
283    }

285    #[test]
286    fn offsets_then_lets_later_edits_replace_earlier_ones() {
287        let src = "ab\ncd\n";
288        let mut first = Rewrite::default();
289        first.keep("ab");
290        first.insert("XY");
291        first.keep("\ncd\n");
292        let (middle, first) = first.finish(src);
293        assert_eq!(middle, "abXY\ncd\n");
294        // Replaces some of what the first put in, and some of the original.
295        let mut second = Rewrite::default();
296        second.keep("abX");
297        second.replace(2, "-");
298        second.keep("cd\n");
299        let (text, second) = second.finish(&middle);
300        assert_eq!(text, "abX-cd\n");
301        let offsets = first.then(second);
302        assert_eq!(offsets.was(src, &text, "ab"), "ab");
303        assert_eq!(offsets.was(src, &text, "X-"), "\n");
304        assert_eq!(offsets.was(src, &text, "-"), "\n");
305        assert_eq!(offsets.was(src, &text, "cd\n"), "cd\n");
306        offsets.assert_fit(src, &text);
307    }
