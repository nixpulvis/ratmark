1use std::borrow::Cow;
2
3use super::{Fence, HtmlTag, Offsets, Rewrite};
4
5/// kramdown reads a line of only a span-level tag, like `<img>` or `<span>`,
6/// as the start of a paragraph that goes on with the lines after it.
7/// CommonMark starts an HTML block there instead, which runs to the next
8/// blank line and leaves any Markdown in it as text. Joining the next line
9/// onto the tag's line makes CommonMark read the same paragraph.
10pub(crate) struct SpanTags;
11
12impl SpanTags {
13    /// The span-level elements kramdown keeps in paragraphs.
14    const NAMES: [&str; 38] = [
15        "a", "abbr", "acronym", "b", "bdo", "big", "br", "button", "cite", "code", "del", "dfn",
16        "em", "i", "img", "input", "ins", "kbd", "label", "mark", "option", "q", "rb", "rbc", "rp",
17        "rt", "rtc", "ruby", "samp", "select", "small", "span", "strong", "sub", "sup", "textarea",
18        "tt", "var",
19    ];
20
21    /// `markdown`, with each line of only a span tag joined onto the next
22    /// line, when that's text, and the offsets of the joins.
23    pub fn rewrite(markdown: &str) -> (Cow<'_, str>, Offsets) {
24        let mut out = Rewrite::of(markdown);
25        let mut fence = Fence::default();
26        let mut lines = markdown.split_inclusive('\n').peekable();
27        while let Some(line) = lines.next() {
28            if fence.read(line) {
29                out.keep(line);
30                continue;
31            }
32            let next_is_text = lines.peek().is_some_and(|next| {
33                let next = next.trim();
34                !next.is_empty() && !next.starts_with("```") && !next.starts_with("~~~")
35            });
36            if next_is_text && Self::alone(line) {
37                let tag = line.trim_end();
38                out.keep(tag);
39                out.replace(line.len() - tag.len(), " ");
40            } else {
41                out.keep(line);
42            }
43        }
44        out.finish(markdown)
45    }
46
47    /// Whether `line` is only a span-level tag, opening or closing, and not
48    /// indented enough to be code.
49    fn alone(line: &str) -> bool {
50        let indent = line.len() - line.trim_start().len();
51        let trimmed = line.trim();
52        let one_tag = trimmed
53            .strip_prefix('<')
54            .and_then(|tag| tag.strip_suffix('>'))
55            .is_some_and(|tag| !tag.contains(['<', '>']));
56        indent < 4
57            && one_tag
58            && HtmlTag::starting(trimmed).is_some_and(|tag| {
59                Self::NAMES
60                    .iter()
61                    .any(|name| name.eq_ignore_ascii_case(tag.name))
62            })
63    }
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69
70    #[test]
71    fn span_tags_rewrite_starts_paragraphs_with_span_tags() {
72        assert_eq!(
73            SpanTags::rewrite("<img src=\"a\" />\n![b](c)\n").0,
74            "<img src=\"a\" /> ![b](c)\n"
75        );
76        assert_eq!(
77            SpanTags::rewrite("<span class=\"g\">\n![a](b)\n![c](d)\n</span>\n").0,
78            "<span class=\"g\"> ![a](b)\n![c](d)\n</span>\n"
79        );
80        // Not block elements, tags before a blank line or a fence, or code.
81        let src = "<div>\n*a*\n</div>\n\n<img />\n\n<b>\n```\n<b>\nx\n```\n    <b>\n    x\n";
82        let (text, offsets) = SpanTags::rewrite(src);
83        assert!(matches!(text, Cow::Borrowed(_)));
84        assert!(offsets.is_empty());
85    }
86
87    #[test]
88    fn span_tags_rewrite_ignores_case_and_joins_closing_tags() {
89        assert_eq!(SpanTags::rewrite("<IMG src=x>\na\n").0, "<IMG src=x> a\n");
90        assert_eq!(SpanTags::rewrite("</span>\na\n").0, "</span> a\n");
91        // Not two tags on a line.
92        assert!(matches!(
93            SpanTags::rewrite("<span><b>\na\n").0,
94            Cow::Borrowed(_)
95        ));
96    }
