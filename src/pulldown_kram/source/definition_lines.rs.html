1use std::borrow::Cow;
2
3use super::{Fence, Offsets, Rewrite};
4use crate::text::Lead;
5
6/// kramdown reads two things about definitions differently from
7/// pulldown-cmark, and this rewrites the lines for both.
8///
9/// - A `:` more than one blank line after the terms starts no definition,
10///   so it's escaped.
11/// - A definition with nothing on its first line has its content indented
12///   by four, rather than two past the `:`. So the lines of it indented by
13///   four or more lose the difference.
14pub(crate) struct DefinitionLines {
15    out: Rewrite,
16    fence: Fence,
17    /// The blank lines in a row before this one.
18    blanks: usize,
19    /// Whether the lines are in a definition. Once they are, a `:` after
20    /// blank lines starts another definition, rather than being escaped.
21    in_definition: bool,
22    /// For a definition with nothing on its first line, the spaces to take
23    /// off its lines, and whether a line of it has been read yet.
24    shift: Option<(usize, bool)>,
25}
26
27impl DefinitionLines {
28    /// `markdown`, with its definitions' lines rewritten as kramdown reads
29    /// them, and the offsets of the edits.
30    pub fn rewrite(markdown: &str) -> (Cow<'_, str>, Offsets) {
31        let starts_definition =
32            |line: &str| line.trim_start_matches([' ', '\t', '>']).starts_with(':');
33        if !markdown.lines().any(starts_definition) {
34            return (Cow::Borrowed(markdown), Offsets::default());
35        }
36        let mut lines = Self {
37            out: Rewrite::of(markdown),
38            fence: Fence::default(),
39            blanks: 0,
40            in_definition: false,
41            shift: None,
42        };
43        for line in markdown.split_inclusive('\n') {
44            lines.line(line);
45        }
46        lines.out.finish(markdown)
47    }
48
49    fn line(&mut self, line: &str) {
50        let shift = self.take_shift(line);
51        if self.fence.read(line) {
52            self.blanks = 0;
53            self.keep(line, shift);
54            return;
55        }
56        let text = line.trim_end_matches(['\n', '\r']);
57        let lead = Lead::of(text);
58        let rest = &text[lead.len..];
59        if rest.trim().is_empty() {
60            self.blanks += 1;
61            self.keep(line, shift);
62            return;
63        }
64        let indent = lead.indent;
65        let marker = indent <= 3
66            && rest
67                .strip_prefix(':')
68                .is_some_and(|after| after.is_empty() || after.starts_with([' ', '\t']));
69        if marker && self.blanks >= 2 && !self.in_definition {
70            self.out.keep(&line[..lead.len]);
71            self.out.insert("\\");
72            self.out.keep(&line[lead.len..]);
73        } else if marker {
74            let first_line_empty = rest[1..].trim().is_empty();
75            let quoted = lead.quoted;
76            if first_line_empty && !quoted && self.blanks == 0 && self.out.read > 0 && indent < 2 {
77                self.shift = Some((2 - indent, false));
78            }
79            self.in_definition = true;
80            self.keep(line, shift);
81        } else {
82            if self.blanks > 0 && indent == 0 {
83                self.in_definition = false;
84            }
85            self.keep(line, shift);
86        }
87        self.blanks = 0;
88    }
89
90    /// The spaces to take off `line`, if it's indented by four or more in a
91    /// definition with nothing on its first line. Such a definition ends at
92    /// a line indented by less than two, or at a blank line right after its
93    /// first.
94    fn take_shift(&mut self, line: &str) -> usize {
95        let Some((by, started)) = self.shift else {
96            return 0;
97        };
98        if line.trim().is_empty() {
99            if !started {
100                self.shift = None;
101            }
102            return 0;
103        }
104        let indent = line.len() - line.trim_start_matches(' ').len();
105        if indent < 2 {
106            self.shift = None;
107            return 0;
108        }
109        self.shift = Some((by, true));
110        if indent >= 4 { by } else { 0 }
111    }
112
113    /// Copies `line`, taking `shift` of the four spaces it starts with off.
114    fn keep(&mut self, line: &str, shift: usize) {
115        if shift == 0 {
116            self.out.keep(line);
117            return;
118        }
119        self.out.replace(4, &"    "[shift..]);
120        self.out.keep(&line[4..]);
121    }
122}

124#[cfg(test)]
125mod tests {
126    use super::*;
127
128    #[test]
129    fn definition_lines_rewrite_escapes_a_definition_far_from_its_terms() {
130        let src = "a\n\n\n: b\n\n> c\n>\n>\n>  : d\n\ne\n\n: f\n\n\n: g\n\n    : h\n";
131        let (text, offsets) = DefinitionLines::rewrite(src);
132        // Not after one blank line, nor after a definition, nor as code.
133        assert_eq!(
134            text,
135            "a\n\n\n\\: b\n\n> c\n>\n>\n>  \\: d\n\ne\n\n: f\n\n\n: g\n\n    : h\n"
136        );
137        assert_eq!(offsets.was(src, &text, "\\: b"), ": b");
138        offsets.assert_fit(src, &text);
139        let (text, _) = DefinitionLines::rewrite("```\na\n\n\n: b\n```\n");
140        assert!(matches!(text, Cow::Borrowed(_)));
141    }
142
143    #[test]
144    fn definition_lines_rewrite_indents_a_definition_empty_on_its_first_line_by_four() {
145        let src = "a\n: \n        b\n   c\n\n      d\ne\n        f\n";
146        let (text, offsets) = DefinitionLines::rewrite(src);
147        // Only the lines indented by four or more, until one by less than two.
148        assert_eq!(text, "a\n: \n      b\n   c\n\n    d\ne\n        f\n");
149        assert_eq!(offsets.was(src, &text, "      b"), "        b");
150        offsets.assert_fit(src, &text);
151        // By one with the `:` indented by one, and not at all after a blank
152        // line, which ends the definition, or with something on the line.
153        for (src, rewritten) in [
154            ("a\n : \n     b\n", "a\n : \n    b\n"),
155            ("a\n: \n\n    b\n", "a\n: \n\n    b\n"),
156            ("a\n: x\n    b\n", "a\n: x\n    b\n"),
157        ] {
158            assert_eq!(DefinitionLines::rewrite(src).0, rewritten, "{src:?}");
159        }
160    }
